1. УСТАНОВКА

pandas можно установить через pip или conda. Для работы с Excel, Parquet, SQL и другими форматами требуются дополнительные зависимости.

Базовые способы установки

conda install -c conda-forge pandas

3. СОЗДАНИЕ ОБЪЕКТОВ
Создание Series
import pandas as pd
import numpy as np

# Из списка
s = pd.Series([10, 20, 30, 40])

# С индексом
s = pd.Series([10, 20, 30], index=['a', 'b', 'c'])

# Из словаря
s = pd.Series({'a': 1, 'b': 2, 'c': 3})

# Из NumPy-массива
s = pd.Series(np.random.randn(5), index=pd.date_range('2026-01-01', periods=5))

# С указанием dtype
s = pd.Series([1, 2, 3], dtype='Int64')
Создание DataFrame
# Из словаря списков
df = pd.DataFrame({
    'name': ['Alice', 'Bob', 'Charlie'],
    'age': [25, 30, 35],
    'city': ['Moscow', 'SPb', 'Kazan']
})

# Из списка словарей
df = pd.DataFrame([
    {'name': 'Alice', 'age': 25},
    {'name': 'Bob', 'age': 30}

df = pd.DataFrame(

4. ИНДЕКСАЦИЯ И ВЫБОРКА

pandas предоставляет несколько способов доступа к данным: [], .loc[], .iloc[], .at[], .iat[]. Наиболее предпочтительны .loc и .iloc.

Разница между .loc и .iloc
МетодПо чемуПример
.loc[]По метке (label-based)df.loc['row1', 'col1']
.iloc[]По позиции (integer-based)df.iloc[0, 1]
.at[]Скаляр по меткеdf.at['row1', 'col1']
.iat[]Скаляр по позицииdf.iat[0, 1]
Примеры выборки
# Столбец (Series)
df['name']
df.name                      # если нет пробелов в имени

# Несколько столбцов (DataFrame)
df[['name', 'age']]

# Срез по строкам
df[2:5]

# Булева маска
df[df['age'] > 30]
df[(df['age'] > 25) & (df['city'] == 'Moscow')]

# .loc — по метке
df.loc['row1']
df.loc['row1':'row5', 'name':'age']
df.loc[df['age'] > 30, ['name', 'age']]

# .iloc — по позиции
df.iloc[0]
df.iloc[0:3, 0:2]
df.iloc[[0, 2, 4], [1, 3]]

# Скаляры (быстро)
df.at['row1', 'name']
df.iat[0, 0]

# Проверка вхождения
df['name'].isin(['Alice', 'Bob'])

# Сортировка
df.sort_values('age', ascending=False)
df.sort_index()
Не используйте цепную индексацию df['a']['b'] = ... — это может привести к SettingWithCopyWarning. Используйте df.loc[...].

6. ПРОПУЩЕННЫЕ ДАННЫЕ

pandas использует NaN (float), NaT (datetime) и pd.NA (nullable-типы) для обозначения отсутствующих значений.

Обнаружение и обработка
# Проверка на пропуски
df.isna()                    # DataFrame булевых значений
df.notna()                   # инверсия
df.isna().sum()              # количество NaN в каждом столбце
df.isna().sum().sum()        # всего NaN

# Удаление
df.dropna()                  # удалить строки с любым NaN
df.dropna(axis=1)            # удалить столбцы с NaN
df.dropna(how='all')         # удалить только если все NaN
df.dropna(thresh=2)          # оставить если ≥2 не-NaN
df.dropna(subset=['age'])    # только по столбцу age

# Заполнение
df.fillna(0)                 # константой
df.fillna(method='ffill')    # вперёд (устарело → df.ffill())
df.ffill()                   # forward fill
df.bfill()                   # backward fill
df['age'].fillna(df['age'].mean())    # средним
df['city'].fillna('Unknown')          # строкой

# Интерполяция
df.interpolate(method='linear')
df.interpolate(method='time')         # для временных рядов
МетодКогда использовать
dropna()Мало пропусков, потеря данных допустима
fillna(value)Значение по умолчанию осмысленно
ffill / bfillВременные ряды, последовательные данные
interpolateЧисловые данные, плавные изменения
mean / medianЧисловые, распределение сохраняется

8. ГРУППИРОВКА (GROUPBY)

Механизм split-apply-combine: разделение данных по ключу, применение агрегации и объединение результата.

Агрегатные функции
ФункцияОписание
sum()Сумма
mean()Среднее
median()Медиана
min() / max()Минимум / максимум
std() / var()Стандартное отклонение / дисперсия
count()Количество значений
size()Количество элементов (с NaN)
nunique()Уникальные значения
first() / last()Первое / последнее
quantile(q)Квантиль
Примеры
# Группировка по одному столбцу
df.groupby('city')['age'].mean()

# Несколько агрегаций
df.groupby('city').agg({
    'age': ['mean', 'max'],
    'salary': 'sum'
})

# Несколько столбцов группировки
df.groupby(['city', 'gender'])['salary'].mean()

# Именованные агрегации (pandas 0.25+)
df.groupby('city').agg(
    avg_age=('age', 'mean'),
    max_salary=('salary', 'max'),
    count=('id', 'count')
)

# Apply — своя функция
df.groupby('city').apply(lambda g: g.nlargest(3, 'salary'))

# Transform — сохраняет форму
df['city_mean_age'] = df.groupby('city')['age'].transform('mean')

# Filter — оставить только группы по условию
df.groupby('city').filter(lambda g: len(g) > 10)

# Cumsum по группам
df['cum_sales'] = df.groupby('city')['sales'].cumsum()

# Сбросить индекс после группировки
result = df.groupby('city').agg(...).reset_index()

# По индексу
pd.merge(df1, df2, left_index=True, right_index=True)

# Указание суффиксов при конфликте имён
pd.merge(df1, df2, on='id', suffixes=('_left', '_right'))

# Проверка уникальности
pd.merge(df1, df2, on='id', validate='one_to_many')

# Индикатор источника строк
pd.merge(df1, df2, on='id', how='outer', indicator=True)

# concat: вертикально
pd.concat([df1, df2], ignore_index=True)

# concat: горизонтально
pd.concat([df1, df2], axis=1)

# concat: с ключами (MultiIndex)
pd.concat([df1, df2], keys=['train', 'test'])

# МЕДЛЕННО (цикл)
for i in range(len(df)):
    df.loc[i, 'y'] = df.loc[i, 'x'] ** 2

# БЫСТРО

# Память: понизить типы
df['a'] = df['a'].astype('int32')
df['b'] = df['b'].astype('float32')
df['c'] = df['c'].astype('category')

# query / eval — быстрее, чем маски
df.query('age > 30 and city == "MSK"')
df.eval('total = price * qty', inplace=True)

# Чтение Parquet с колонками (columnar pushdown)

# Arrow backend

import pandas as pd

def etl_pipeline(input_path, output_path):
    # 1. Чтение
    df = pd.read_csv(
        input_path,
        parse_dates=['date'],
        dtype={'user_id': 'int64', 'amount': 'float64'},
        dtype_backend='pyarrow'
    )
    
    # 2. Очистка
    df = df.dropna(subset=['user_id', 'amount'])
    df = df[df['amount'] > 0]
    df['amount'] = df['amount'].clip(upper=1_000_000)
    df['date'] = pd.to_datetime(df['date'], errors='coerce')
    df = df.dropna(subset=['date'])
    
    # 3. Обогащение
    df['year'] = df['date'].dt.year
    df['month'] = df['date'].dt.to_period('M')
    df['weekday'] = df['date'].dt.day_name()
    
    # 4. Агрегация
    result = df.groupby(['user_id', 'month']).agg(
        total=('amount', 'sum'),
        count=('amount', 'count'),
        avg=('amount', 'mean'),
        max=('amount', 'max')
    ).reset_index()
    
    # 5. Запись
    result.to_parquet(output_path, index=False)
    return result

df = etl_pipeline('input.csv', 'output.parquet')
print(df.head())
Анализ временного ряда
import pandas as pd
import numpy as np

# Генерация синтетики
dates = pd.date_range('2024-01-01', '2026-10-01', freq='D')
df = pd.DataFrame({
    'date': dates,
    'sales': np.random.poisson(100, len(dates)) + 
             np.sin(np.arange(len(dates)) / 30) * 20
})
df = df.set_index('date')

# Скользящие метрики
df['ma7'] = df['sales'].rolling(7).mean()
df['ma30'] = df['sales'].rolling(30).mean()
df['std30'] = df['sales'].rolling(30).std()

# Аномалии (выход за 3 сигмы)
df['z_score'] = (df['sales'] - df['ma30']) / df['std30']
df['anomaly'] = df['z_score'].abs() > 3

# Месячная агрегация
monthly = df.resample('M').agg({
    'sales': ['sum', 'mean', 'std', 'max'],
    'anomaly': 'sum'
})
print(monthly.tail())
Объединение данных из нескольких источников
import pandas as pd

# Чтение
users = pd.read_csv('users.csv')
orders = pd.read_parquet('orders.parquet')
products = pd.read_excel('products.xlsx')

# Проверка ключей
assert users['user_id'].is_unique, "user_id не уникален!"
assert products['product_id'].is_unique

# Объединение orders + users
df = orders.merge(
    users[['user_id', 'name', 'city']],
    on='user_id',
    how='left',
    validate='many_to_one'
)

# Объединение с products
df = df.merge(
    products[['product_id', 'category', 'price']],
    on='product_id',
    how='left'
)

# Расчёт revenue
df['revenue'] = df['qty'] * df['price']

# Топ-10 клиентов по выручке
top = (df.groupby(['user_id', 'name'])
         .agg(revenue=('revenue', 'sum'),
              orders=('order_id', 'count'))
         .sort_values('revenue', ascending=False)
         .head(10))
print(top)
