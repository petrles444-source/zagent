1. УСТАНОВКА

OpenCV можно установить тремя основными способами: через менеджер пакетов Python (pip), системный менеджер пакетов (apt, brew) или собрать из исходников для максимальной производительности.

Python (pip)
# Базовая установка
pip install opencv-python

# С contrib-модулями (SIFT, SURF, трекеры)
pip install opencv-contrib-python

# Для headless-серверов (без GUI)
pip install opencv-python-headless

# Проверка установки
python -c "import cv2; print(cv2.__version__)"
Linux (Ubuntu / Debian)
sudo apt update
sudo apt install -y libopencv-dev python3-opencv

# Проверка
pkg-config --modversion opencv4
Сборка из исходников (CMake)

Для сборки из исходников требуется C++ компилятор (GCC/G++ или Clang), CMake 3.14+, а также зависимости для работы с изображениями и видео.

# 1. Установка зависимостей (Ubuntu)
sudo apt install -y build-essential cmake git pkg-config \
  libgtk-3-dev libavcodec-dev libavformat-dev libswscale-dev \
  libjpeg-dev libpng-dev libtiff-dev

# 2. Клонирование
git clone https://github.com/opencv/opencv.git
git clone https://github.com/opencv/opencv_contrib.git

# 3. Конфигурация
cd opencv && mkdir build && cd build
cmake -D CMAKE_BUILD_TYPE=Release \
      -D CMAKE_INSTALL_PREFIX=/usr/local \
      -D OPENCV_EXTRA_MODULES_PATH=../../opencv_contrib/modules \
      ..

# 4. Сборка (используйте все ядра)
make -j$(nproc)
sudo make install
Сборка из исходников даёт максимальную производительность и позволяет включить CUDA, OpenCL, TBB и другие оптимизации.

Пример: чтение, отображение, запись
import cv2

# Чтение изображения в цвете (BGR)
img = cv2.imread('input.jpg', cv2.IMREAD_COLOR)

# Проверка успешности загрузки
if img is None:
    print("Ошибка: не удалось загрузить изображение")
    exit()

# Получение размеров
height, width, channels = img.shape
print(f"Размер: {width}x{height}, каналов: {channels}")

# Отображение в окне
cv2.imshow('Preview', img)
cv2.waitKey(0)          # Ждать нажатия клавиши
cv2.destroyAllWindows()

# Сохранение в файл
cv2.imwrite('output.png', img)
По умолчанию OpenCV читает изображения в формате BGR (Blue-Green-Red), а не RGB. Это важно помнить при интеграции с другими библиотеками (matplotlib, PIL).
4. ЦВЕТОВЫЕ ПРОСТРАНСТВА

OpenCV поддерживает более 150 цветовых пространств. Наиболее часто используются BGR, Grayscale, HSV и Lab. Конвертация выполняется функцией cv2.cvtColor()[reference:5].

Основные коды конвертации

COLOR_BGR2HSVBGR → HSV (для цветовой сегментации)
COLOR_BGR2LABBGR → Lab (перцептуальное пространство)

Пример: конвертация и разделение каналов
import cv2

img = cv2.imread('input.jpg')

# Конвертация в grayscale
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# Конвертация в HSV
hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

# Разделение на каналы
b, g, r = cv2.split(img)

# Объединение каналов обратно
merged = cv2.merge([b, g, r])

# Цветовая сегментация (поиск красных объектов)
lower_red = (0, 100, 100)
upper_red = (10, 255, 255)
mask = cv2.inRange(hsv, lower_red, upper_red)
result = cv2.bitwise_and(img, img, mask=mask)

import cv2

7. ДЕТЕКЦИЯ КРАЁВ

Детекция краёв — одна из базовых задач компьютерного зрения. Наиболее популярные методы: Canny, Sobel, Laplacian, Scharr.

Оператор Canny
import cv2

img = cv2.imread('input.jpg', cv2.IMREAD_GRAYSCALE)

# Canny: порог1, порог2
edges = cv2.Canny(img, 100, 200)

# С автоматическим подбором порогов
median = np.median(img)
lower = int(max(0, 0.66 * median))
upper = int(min(255, 1.33 * median))
auto_edges = cv2.Canny(img, lower, upper)
Оператор Sobel и Laplacian
# Градиент по X
sobelx = cv2.Sobel(img, cv2.CV_64F, 1, 0, ksize=5)
# Градиент по Y
sobely = cv2.Sobel(img, cv2.CV_64F, 0, 1, ksize=5)

# Объединение градиентов
sobel = cv2.magnitude(sobelx, sobely)

# Laplacian
laplacian = cv2.Laplacian(img, cv2.CV_64F)

# Приведение к 8-битному формату
sobel_abs = cv2.convertScaleAbs(sobel)
laplacian_abs = cv2.convertScaleAbs(laplacian)
Детекция прямых: преобразование Хафа
# Стандартное преобразование Хафа
lines = cv2.HoughLines(edges, 1, np.pi/180, 200)

# Вероятностное преобразование Хафа
lines_p = cv2.HoughLinesP(edges, 1, np.pi/180, 100,
                           minLineLength=100, maxLineGap=10)

# Отрисовка линий
for line in lines_p:
    x1, y1, x2, y2 = line[0]
    cv2.line(img, (x1, y1), (x2, y2), (0, 255, 0), 2)
8. КОНТУРЫ

Контуры — это кривые, соединяющие непрерывные точки с одинаковым цветом или интенсивностью. Используются для анализа форм и детекции объектов.

Поиск и отрисовка контуров
import cv2

img = cv2.imread('input.jpg')
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# Бинаризация (Otsu)
_, thresh = cv2.threshold(gray, 0, 255, 
                          cv2.THRESH_BINARY + cv2.THRESH_OTSU)

# Поиск контуров
contours, hierarchy = cv2.findContours(
    thresh, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)

# Отрисовка всех контуров
cv2.drawContours(img, contours, -1, (0, 255, 0), 2)

# Анализ каждого контура
for c in contours:
    # Площадь
    area = cv2.contourArea(c)
    
    # Периметр
    perimeter = cv2.arcLength(c, True)
    
    # Аппроксимация полигоном (точность 2% от периметра)
    approx = cv2.approxPolyDP(c, 0.02 * perimeter, True)
    
    # Описанный прямоугольник
    x, y, w, h = cv2.boundingRect(c)
    
    # Описанная окружность
    (cx, cy), radius = cv2.minEnclosingCircle(c)
    
    # Выпуклая оболочка
    hull = cv2.convexHull(c)
    
    # Проверка выпуклости
    is_convex = cv2.isContourConvex(c)
Флаги поиска: RETR_EXTERNAL (только внешние), RETR_LIST (все без иерархии), RETR_TREE (полная иерархия), RETR_CCOMP (двухуровневая).
9. ОСОБЫЕ ТОЧКИ (FEATURES2D)

Особые точки используются для сопоставления изображений, трекинга, панорамной склейки и распознавания объектов.

Популярные детекторы
ДетекторТипОсобенности
ORBBinaryБыстрый, бесплатный, хорош для реального времени
SIFTFloatИнвариантен к масштабу и повороту (в contrib с 4.4.0)
SURFFloatБыстрее SIFT (запатентован)

BRISKBinaryБыстрый, инвариантен к масштабу
KAZEFloatНелинейное масштабное пространство
Пример: ORB + BFMatcher
import cv2

img1 = cv2.imread('box.png', cv2.IMREAD_GRAYSCALE)
img2 = cv2.imread('box_in_scene.png', cv2.IMREAD_GRAYSCALE)

# Создание ORB-детектора
orb = cv2.ORB_create(nfeatures=500)

# Поиск ключевых точек и дескрипторов
kp1, des1 = orb.detectAndCompute(img1, None)
kp2, des2 = orb.detectAndCompute(img2, None)

# BFMatcher с расстоянием Хэмминга
bf = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True)
matches = bf.match(des1, des2)

# Сортировка по расстоянию
matches = sorted(matches, key=lambda x: x.distance)

# Отрисовка первых 30 совпадений
result = cv2.drawMatches(img1, kp1, img2, kp2, 
                          matches[:30], None, flags=2)
cv2.imshow('Matches', result)
cv2.waitKey(0)
10. РАБОТА С ВИДЕО

Модуль videoio предоставляет унифицированный интерфейс для захвата и записи видео с различных бэкендов: FFmpeg, GStreamer, DirectShow, AVFoundation[reference:8].

Захват видео с камеры
import cv2

# Открытие камеры (0 — камера по умолчанию)
cap = cv2.VideoCapture(0)

# Проверка открытия
if not cap.isOpened():
    print("Ошибка: не удалось открыть камеру")
    exit()

# Настройка разрешения
cap.set(cv2.CAP_PROP_FRAME_WIDTH, 1280)
cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 720)

# Чтение кадров в цикле
while True:
    ret, frame = cap.read()
    if not ret:
        print("Не удалось получить кадр")
        break
    
    # Обработка кадра
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    cv2.imshow('Frame', gray)
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
Запись видео в файл
import cv2

cap = cv2.VideoCapture(0)

# Определение кодека и создание VideoWriter
fourcc = cv2.VideoWriter_fourcc(*'XVID')
out = cv2.VideoWriter('output.avi', fourcc, 20.0, (640, 480))

while cap.isOpened():
    ret, frame = cap.read()
    if not ret:
        break
    
    out.write(frame)
    cv2.imshow('Recording', frame)
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
out.release()
cv2.destroyAllWindows()
Чтение видео из файла
cap = cv2.VideoCapture('input.mp4')

11. DEEP LEARNING (DNN)

Модуль dnn позволяет загружать и запускать предобученные нейронные сети из популярных фреймворков: TensorFlow, Caffe, ONNX, Darknet.

Загрузка модели
import cv2

# Загрузка ONNX-модели
net = cv2.dnn.readNetFromONNX('model.onnx')

# Загрузка TensorFlow
net = cv2.dnn.readNetFromTensorflow('frozen_graph.pb', 'config.pbtxt')

# Загрузка Caffe
net = cv2.dnn.readNetFromCaffe('deploy.prototxt', 'weights.caffemodel')

# Загрузка Darknet (YOLO)

# Использование GPU (CUDA)
net.setPreferableBackend(cv2.dnn.DNN_BACKEND_CUDA)
net.setPreferableTarget(cv2.dnn.DNN_TARGET_CUDA)
Инференс (пример YOLO)
import cv2
import numpy as np

# Загрузка YOLO
net = cv2.dnn.readNetFromDarknet('yolov3.cfg', 'yolov3.weights')

# Получение выходных слоёв
layer_names = net.getLayerNames()
output_layers = [layer_names[i - 1] for i in net.getUnconnectedOutLayers()]

# Подготовка изображения
img = cv2.imread('input.jpg')
blob = cv2.dnn.blobFromImage(img, 1/255.0, (416, 416), 
                              swapRB=True, crop=False)
net.setInput(blob)

# Прямой проход
outputs = net.forward(output_layers)

# Парсинг результатов
for output in outputs:
    for detection in output:
        scores = detection[5:]
        class_id = np.argmax(scores)
        confidence = scores[class_id]
        
        if confidence > 0.5:
            # Координаты
            center_x = int(detection[0] * img.shape[1])
            center_y = int(detection[1] * img.shape[0])
            w = int(detection[2] * img.shape[1])
            h = int(detection[3] * img.shape[0])
            
            # Прямоугольник
            x = int(center_x - w / 2)
            y = int(center_y - h / 2)
            cv2.rectangle(img, (x, y), (x+w, y+h), (0, 255, 0), 2)
12. РИСОВАНИЕ И GUI

Модуль highgui предоставляет функции для отображения изображений в окнах, а также для рисования геометрических фигур и текста[reference:9].

Рисование фигур
import cv2
import numpy as np

# Создание пустого изображения (чёрный фон)
img = np.zeros((512, 512, 3), np.uint8)

# Линия (изображение, начало, конец, цвет BGR, толщина)
cv2.line(img, (0, 0), (511, 511), (255, 0, 0), 5)

# Прямоугольник (изображение, угол1, угол2, цвет, толщина)
cv2.rectangle(img, (384, 0), (510, 128), (0, 255, 0), 3)

# Окружность (изображение, центр, радиус, цвет, толщина)
cv2.circle(img, (447, 63), 63, (0, 0, 255), -1)

# Эллипс (изображение, центр, оси, угол, startAngle, endAngle, цвет)
cv2.ellipse(img, (256, 256), (100, 50), 0, 0, 180, 
            (255, 255, 255), -1)

# Полигон
pts = np.array([[10, 5], [20, 30], [70, 20], [50, 10]], np.int32)
pts = pts.reshape((-1, 1, 2))

Полный пайплайн обработки изображения
import cv2
import numpy as np

def process_image(path):
    # 1. Чтение
    img = cv2.imread(path)
    if img is None:
        raise FileNotFoundError(path)
    
    # 2. Ресайз до 640x480
    img = cv2.resize(img, (640, 480))
    
    # 3. Конвертация в grayscale
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    
    # 4. Шумоподавление
    blur = cv2.GaussianBlur(gray, (5, 5), 0)
    
    # 5. Бинаризация
    _, thresh = cv2.threshold(blur, 0, 255, 
                              cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    
    # 6. Детекция краёв
    edges = cv2.Canny(thresh, 50, 150)
    
    # 7. Морфология (закрытие)
    kernel = np.ones((3, 3), np.uint8)
    closed = cv2.morphologyEx(edges, cv2.MORPH_CLOSE, kernel)
    
    return img, gray, thresh, closed

# Использование
original, gray, binary, edges = process_image('input.jpg')
cv2.imshow('Original', original)
cv2.imshow('Edges', edges)
cv2.waitKey(0)
Детекция лиц через Haar Cascade
import cv2

# Загрузка каскада (входит в OpenCV)
face_cascade = cv2.CascadeClassifier(
    cv2.data.haarcascades + 'haarcascade_frontalface_default.xml')

# Захват видео
cap = cv2.VideoCapture(0)

while True:
    ret, frame = cap.read()
    if not ret:
        break
    
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    # Детекция лиц
    faces = face_cascade.detectMultiScale(
        gray, scaleFactor=1.1, minNeighbors=5, minSize=(30, 30))
    
    # Отрисовка прямоугольников
    for (x, y, w, h) in faces:
        cv2.rectangle(frame, (x, y), (x+w, y+h), (0, 255, 0), 2)
        cv2.putText(frame, 'Face', (x, y-10),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.9, (0, 255, 0), 2)
    
    cv2.imshow('Faces', frame)
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()

import cv2
import numpy as np

cap = cv2.VideoCapture('input.mp4')

# Получение свойств
fps = cap.get(cv2.CAP_PROP_FPS)
w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
total = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))

# VideoWriter
fourcc = cv2.VideoWriter_fourcc(*'mp4v')
out = cv2.VideoWriter('output.mp4', fourcc, fps, (w, h))

frame_num = 0
while True:
    ret, frame = cap.read()
    if not ret:
        break
    
    frame_num += 1
    
    # Обработка кадра
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    edges = cv2.Canny(gray, 50, 150)
    result = cv2.cvtColor(edges, cv2.COLOR_GRAY2BGR)
    
    # Запись
    out.write(result)
    
    # Прогресс
    if frame_num % 30 == 0:
        print(f"Обработано: {frame_num}/{total}")

cap.release()
out.release()
print("Готово!")
