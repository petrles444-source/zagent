# ---------- Алфавит и токенизация ----------
RUS_ALPHABET = "абвгдеёжзийклмнопрстуфхцчшщъыьэюя"

def normalize_text(text: str, lower: bool = True, keep_yo: bool = True) -> str:
    # Unicode NFC
    import unicodedata
    text = unicodedata.normalize("NFC", text)
    if lower:
        text = text.lower()
    if not keep_yo:
        text = text.replace("ё", "е")
    # Можно заменить множественные пробелы, но для char-LM не обязательно
    return text

def build_vocab(text: str, alphabet: str = RUS_ALPHABET) -> Tuple[List[str], Dict[str, int], Dict[int, str]]:
    # Спецтокены
    specials = ["<pad>", "<unk>", "<bos>", "<eos>"]
    # Гарантируем наличие русских букв
    chars = list(alphabet)
    # Добавляем остальные символы из текста
    for ch in text:
        if ch not in chars:
            chars.append(ch)
    # Итоговый словарь: specials + chars
    itos = specials + chars
    stoi = {ch: i for i, ch in enumerate(itos)}
    return itos, stoi, {i: ch for ch, i in stoi.items()}

# ---------- Dataset ----------
class CharDataset(Dataset):
    def __init__(self, data: torch.Tensor, block_size: int):
        self.data = data
        self.block_size = block_size
    def __len__(self):
        return len(self.data) - self.block_size
    def __getitem__(self, idx):
        x = self.data[idx:idx+self.block_size]
        y = self.data[idx+1:idx+self.block_size+1]
        return x, y

# ---------- Модель ----------
class CharLSTM(nn.Module):
    def __init__(self, vocab_size, emb_dim, hidden_dim, num_layers, dropout):
        super().__init__()
        self.emb = nn.Embedding(vocab_size, emb_dim)
        self.lstm = nn.LSTM(emb_dim, hidden_dim, num_layers, batch_first=True, dropout=dropout if num_layers>1 else 0.0)
        self.drop = nn.Dropout(dropout)
        self.fc = nn.Linear(hidden_dim, vocab_size)
    def forward(self, x, hidden=None):
        emb = self.emb(x)  # (B, T, E)
        out, hidden = self.lstm(emb, hidden)
        out = self.drop(out)
        logits = self.fc(out)  # (B, T, V)
        return logits, hidden

# ---------- Обучение ----------
def train_epoch(model, loader, optimizer, criterion, cfg):
    model.train()
    total_loss = 0
    for x, y in loader:
        x, y = x.to(cfg.device), y.to(cfg.device)
        optimizer.zero_grad()
        logits, _ = model(x)
        loss = criterion(logits.view(-1, logits.size(-1)), y.view(-1))
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), cfg.grad_clip)
        optimizer.step()
        total_loss += loss.item()
    return total_loss / len(loader)

@torch.no_grad()
def evaluate(model, loader, criterion, cfg):
    model.eval()
    total_loss = 0
    for x, y in loader:
        x, y = x.to(cfg.device), y.to(cfg.device)
        logits, _ = model(x)
        loss = criterion(logits.view(-1, logits.size(-1)), y.view(-1))
        total_loss += loss.item()
    avg_loss = total_loss / len(loader)
    perplexity = math.exp(avg_loss)
    return avg_loss, perplexity

# ---------- Генерация ----------
@torch.no_grad()
def generate(model, stoi, itos, start_str="", max_new_tokens=200, temperature=0.8, top_k=10, cfg=cfg):
    model.eval()
    # Если start_str пуст, начинаем с <bos>
    if start_str:
        ids = [stoi.get(ch, stoi["<unk>"]) for ch in start_str]
    else:
        ids = [stoi["<bos>"]]
    x = torch.tensor(ids, dtype=torch.long, device=cfg.device).unsqueeze(0)  # (1, T)
    generated = list(ids)
    hidden = None
    for _ in range(max_new_tokens):
        # Обрезаем до block_size
        x_cond = x[:, -cfg.block_size:]
        logits, hidden = model(x_cond, hidden)
        logits = logits[:, -1, :] / temperature  # (1, V)
        # top-k
        if top_k is not None:
            v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
            logits[logits < v[:, [-1]]] = -float('inf')
        probs = F.softmax(logits, dim=-1)
        next_id = torch.multinomial(probs, num_samples=1)  # (1,1)
        generated.append(next_id.item())
        x = torch.cat([x, next_id], dim=1)
        if next_id.item() == stoi.get("<eos>", -1):
            break
    # Декодируем, убирая спецтокены
    chars = []
    for i in generated:
        ch = itos[i]
        if ch in ("<pad>", "<bos>", "<eos>"):
            continue
        if ch == "<unk>":
            ch = "?"
        chars.append(ch)
    return "".join(chars)

import os
import math
import random
import unicodedata
from dataclasses import dataclass
from typing import List, Tuple, Dict

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import Dataset, DataLoader


# =========================
# Конфигурация
# =========================
@dataclass
class Config:
    data_path: str = "data.txt"
    block_size: int = 128          # длина контекста
    batch_size: int = 64
    emb_dim: int = 256
    hidden_dim: int = 512
    num_layers: int = 2
    dropout: float = 0.2
    lr: float = 3e-4
    weight_decay: float = 1e-4
    epochs: int = 20
    grad_clip: float = 1.0
    seed: int = 42
    device: str = "cuda" if torch.cuda.is_available() else "cpu"
    val_ratio: float = 0.1
    save_path: str = "best_char_lm.pt"


cfg = Config()


# =========================
# Русский алфавит и токенизация
# =========================
RUS_ALPHABET = "абвгдеёжзийклмнопрстуфхцчшщъыьэюя"


def normalize_text(text: str, lower: bool = True, keep_yo: bool = True) -> str:
    """
    Приводит текст к удобному виду:
    - Unicode NFC
    - нижний регистр
    - при желании заменяет ё -> е
    """
    text = unicodedata.normalize("NFC", text)
    if lower:
        text = text.lower()
    if not keep_yo:
        text = text.replace("ё", "е")
    return text


def build_vocab(text: str, alphabet: str = RUS_ALPHABET):
    """
    Строит словарь символ -> индекс.
    Спецтокены идут первыми, затем русский алфавит,
    затем все остальные символы, встреченные в тексте.
    """
    specials = ["<pad>", "<unk>", "<bos>", "<eos>"]
    chars = list(alphabet)

    for ch in text:
        if ch not in chars:
            chars.append(ch)

    itos = specials + chars
    stoi = {ch: i for i, ch in enumerate(itos)}
    idx2char = {i: ch for ch, i in stoi.items()}
    return itos, stoi, idx2char


# =========================
# Dataset
# =========================
class CharDataset(Dataset):
    """
    x = последовательность символов длины block_size
    y = та же последовательность, сдвинутая на 1 вправо
    """
    def __init__(self, data: torch.Tensor, block_size: int):
        self.data = data
        self.block_size = block_size

    def __len__(self):
        return max(0, len(self.data) - self.block_size)

    def __getitem__(self, idx):
        x = self.data[idx: idx + self.block_size]
        y = self.data[idx + 1: idx + self.block_size + 1]
        return x, y


# =========================
# Модель: LSTM для предсказания следующего символа
# =========================
class CharLSTM(nn.Module):
    def __init__(self, vocab_size, emb_dim, hidden_dim, num_layers, dropout):
        super().__init__()
        self.emb = nn.Embedding(vocab_size, emb_dim)
        self.lstm = nn.LSTM(
            emb_dim,
            hidden_dim,
            num_layers,
            batch_first=True,
            dropout=dropout if num_layers > 1 else 0.0,
        )
        self.drop = nn.Dropout(dropout)
        self.fc = nn.Linear(hidden_dim, vocab_size)

    def forward(self, x, hidden=None):
        # x: (B, T)
        emb = self.emb(x)                 # (B, T, E)
        out, hidden = self.lstm(emb, hidden)
        out = self.drop(out)
        logits = self.fc(out)             # (B, T, V)
        return logits, hidden


# =========================
# Обучение и валидация
# =========================
def train_epoch(model, loader, optimizer, criterion, cfg):
    model.train()
    total_loss = 0.0

    for x, y in loader:
        x, y = x.to(cfg.device), y.to(cfg.device)

        optimizer.zero_grad()
        logits, _ = model(x)

        loss = criterion(
            logits.view(-1, logits.size(-1)),
            y.view(-1)
        )
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), cfg.grad_clip)
        optimizer.step()

        total_loss += loss.item()

    return total_loss / max(1, len(loader))


@torch.no_grad()
def evaluate(model, loader, criterion, cfg):
    model.eval()
    total_loss = 0.0

    for x, y in loader:
        x, y = x.to(cfg.device), y.to(cfg.device)
        logits, _ = model(x)
        loss = criterion(
            logits.view(-1, logits.size(-1)),
            y.view(-1)
        )
        total_loss += loss.item()

    avg_loss = total_loss / max(1, len(loader))
    perplexity = math.exp(avg_loss)
    return avg_loss, perplexity


# =========================
# Генерация текста
# =========================
@torch.no_grad()
def generate(
    model,
    stoi,
    itos,
    start_str="",
    max_new_tokens=200,
    temperature=0.8,
    top_k=10,
    cfg=cfg,
):
    model.eval()

    if start_str:
        ids = [stoi.get(ch, stoi["<unk>"]) for ch in start_str]
    else:
        ids = [stoi["<bos>"]]

    x = torch.tensor(ids, dtype=torch.long, device=cfg.device).unsqueeze(0)
    generated = list(ids)

    for _ in range(max_new_tokens):
        x_cond = x[:, -cfg.block_size:]
        logits, _ = model(x_cond)
        logits = logits[:, -1, :] / temperature

        if top_k is not None:
            v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
            logits[logits < v[:, [-1]]] = -float("inf")

        probs = F.softmax(logits, dim=-1)
        next_id = torch.multinomial(probs, num_samples=1)

        generated.append(next_id.item())
        x = torch.cat([x, next_id], dim=1)

        if next_id.item() == stoi.get("<eos>", -1):
            break

    chars = []
    for i in generated:
        ch = itos[i]
        if ch in ("<pad>", "<bos>", "<eos>"):
            continue
        if ch == "<unk>":
            ch = "?"
        chars.append(ch)

    return "".join(chars)


# =========================
# Main
# =========================
def main():
    random.seed(cfg.seed)
    torch.manual_seed(cfg.seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(cfg.seed)

    # 1. Загрузка текста
    if os.path.exists(cfg.data_path):
        with open(cfg.data_path, "r", encoding="utf-8") as f:
            text = f.read()
        print(f"Загружен файл: {cfg.data_path}")
    else:
        text = """
        Жил-был на свете маленький нейросетевой кот. Он очень любил русский язык,
        буквы, слова и запах свежих данных. Каждое утро он просыпался и начинал
        учиться предсказывать следующую букву. Сначала у него получалось плохо:
        он путал «ё» и «е», «ъ» и «ь». Но постепенно он становился всё умнее.
        Однажды он написал: «Привет, мир! Я умею генерировать текст по буквам».
        И все вокруг удивились.
        """.strip()
        print("Файл data.txt не найден, использую встроенный пример.")

    text = normalize_text(text, lower=True, keep_yo=True)
    print(f"Длина текста: {len(text)} символов")

    # 2. Словарь
    itos, stoi, idx2char = build_vocab(text)
    vocab_size = len(itos)
    print(f"Размер словаря: {vocab_size}")

    # 3. Кодирование
    ids = (
        [stoi["<bos>"]]
        + [stoi.get(ch, stoi["<unk>"]) for ch in text]
        + [stoi["<eos>"]]
    )
    data = torch.tensor(ids, dtype=torch.long)
    print(f"Всего токенов: {len(data)}")

    if len(data) <= cfg.block_size + 1:
        raise ValueError("Текст слишком короткий для выбранного block_size.")

    # 4. Train / Val split
    n = len(data)
    n_val = int(n * cfg.val_ratio)
    n_train = n - n_val

    train_data = data[:n_train]
    val_data = data[n_train:]

    train_dataset = CharDataset(train_data, cfg.block_size)
    val_dataset = CharDataset(val_data, cfg.block_size)

    train_loader = DataLoader(
        train_dataset,
        batch_size=cfg.batch_size,
        shuffle=True,
        drop_last=True,
    )
    val_loader = DataLoader(
        val_dataset,
        batch_size=cfg.batch_size,
        shuffle=False,
        drop_last=False,
    )

    # 5. Модель, оптимизатор, лосс
    model = CharLSTM(
        vocab_size=vocab_size,
        emb_dim=cfg.emb_dim,
        hidden_dim=cfg.hidden_dim,
        num_layers=cfg.num_layers,
        dropout=cfg.dropout,
    ).to(cfg.device)

    print(model)

    optimizer = torch.optim.AdamW(
        model.parameters(),
        lr=cfg.lr,
        weight_decay=cfg.weight_decay,
    )
    criterion = nn.CrossEntropyLoss()

    # 6. Цикл обучения
    best_val_loss = float("inf")

    for epoch in range(1, cfg.epochs + 1):
        train_loss = train_epoch(model, train_loader, optimizer, criterion, cfg)
        val_loss, ppl = evaluate(model, val_loader, criterion, cfg)

        print(
            f"Epoch {epoch:02d} | "
            f"train_loss {train_loss:.4f} | "
            f"val_loss {val_loss:.4f} | "
            f"ppl {ppl:.2f}"
        )

        if val_loss < best_val_loss:
            best_val_loss = val_loss
            torch.save(
                {
                    "model_state": model.state_dict(),
                    "itos": itos,
                    "stoi": stoi,
                    "cfg": cfg.__dict__,
                },
                cfg.save_path,
            )
            print(f"  -> сохранено в {cfg.save_path}")

    # 7. Генерация примеров
    print("\n--- Генерация ---")
    for prompt in ["", "нейросет", "привет"]:
        out = generate(
            model,
            stoi,
            itos,
            start_str=prompt,
            max_new_tokens=200,
            temperature=0.8,
            top_k=10,
            cfg=cfg,
        )
        print(f"Промпт: {prompt!r}\n{out}\n")


if __name__ == "__main__":
    main()

    def forward(self, x, hidden=None):
        emb = self.emb(x)
        out, hidden = self.lstm(emb, hidden)
        out = self.drop(out)
        logits = self.fc(out)
        return logits, hidden


# =====================================================================
# Циклы обучения / валидации
# =====================================================================
def train_epoch(model, loader, optimizer, criterion, cfg):
    model.train()
    total = 0.0
    for x, y in loader:
        x, y = x.to(cfg.device), y.to(cfg.device)
        optimizer.zero_grad()
        logits, _ = model(x)
        loss = criterion(logits.view(-1, logits.size(-1)), y.view(-1))
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), cfg.grad_clip)
        optimizer.step()
        total += loss.item()
    return total / max(1, len(loader))


@torch.no_grad()
def evaluate(model, loader, criterion, cfg):
    model.eval()
    total = 0.0
    for x, y in loader:
        x, y = x.to(cfg.device), y.to(cfg.device)
        logits, _ = model(x)
        loss = criterion(logits.view(-1, logits.size(-1)), y.view(-1))
        total += loss.item()
    avg = total / max(1, len(loader))
    return avg, math.exp(avg)


# =====================================================================
# Main
# =====================================================================
def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--data", type=str, default="data.txt")
    p.add_argument("--save", type=str, default="best_char_lm.pt")
    p.add_argument("--epochs", type=int, default=20)
    p.add_argument("--batch-size", type=int, default=64)
    p.add_argument("--block-size", type=int, default=128)
    p.add_argument("--lr", type=float, default=3e-4)
    p.add_argument("--emb-dim", type=int, default=256)
    p.add_argument("--hidden-dim", type=int, default=512)
    p.add_argument("--num-layers", type=int, default=2)
    p.add_argument("--dropout", type=float, default=0.2)
    return p.parse_args()


def main():
    args = parse_args()

    cfg = Config(
        data_path=args.data,
        save_path=args.save,
        epochs=args.epochs,
        batch_size=args.batch_size,
        block_size=args.block_size,
        lr=args.lr,
        emb_dim=args.emb_dim,
        hidden_dim=args.hidden_dim,
        num_layers=args.num_layers,
        dropout=args.dropout,
    )

    random.seed(cfg.seed)
    torch.manual_seed(cfg.seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(cfg.seed)

    print(f"Устройство: {cfg.device}")

    # 1. Загрузка текста
    if not os.path.exists(cfg.data_path):
        raise FileNotFoundError(
            f"Не найден файл {cfg.data_path}. Положите рядом русский текст."
        )
    with open(cfg.data_path, "r", encoding="utf-8") as f:
        text = f.read()

    text = normalize_text(text, lower=True, keep_yo=True)
    print(f"Длина текста: {len(text)} символов")

    # 2. Словарь
    itos, stoi = build_vocab(text)
    vocab_size = len(itos)
    print(f"Размер словаря: {vocab_size}")

    # 3. Кодирование
    ids = (
        [stoi["<bos>"]]
        + [stoi.get(ch, stoi["<unk>"]) for ch in text]
        + [stoi["<eos>"]]
    )
    data = torch.tensor(ids, dtype=torch.long)
    print(f"Всего токенов: {len(data)}")

    if len(data) <= cfg.block_size + 1:
        raise ValueError("Текст слишком короткий для выбранного block_size.")

    # 4. Train/Val
    n = len(data)
    n_val = int(n * cfg.val_ratio)
    n_train = n - n_val

    train_data = data[:n_train]
    val_data = data[n_train:]

    train_loader = DataLoader(
        CharDataset(train_data, cfg.block_size),
        batch_size=cfg.batch_size,
        shuffle=True,
        drop_last=True,
    )
    val_loader = DataLoader(
        CharDataset(val_data, cfg.block_size),
        batch_size=cfg.batch_size,
        shuffle=False,
        drop_last=False,
    )

    # 5. Модель
    model = CharLSTM(
        vocab_size=vocab_size,
        emb_dim=cfg.emb_dim,
        hidden_dim=cfg.hidden_dim,
        num_layers=cfg.num_layers,
        dropout=cfg.dropout,
    ).to(cfg.device)

    print(model)
    n_params = sum(p.numel() for p in model.parameters())
    print(f"Параметров: {n_params:,}")

    optimizer = torch.optim.AdamW(
        model.parameters(), lr=cfg.lr, weight_decay=cfg.weight_decay
    )
    criterion = nn.CrossEntropyLoss()

    # 6. Обучение
    best_val = float("inf")

    for epoch in range(1, cfg.epochs + 1):
        tr = train_epoch(model, train_loader, optimizer, criterion, cfg)
        vl, ppl = evaluate(model, val_loader, criterion, cfg)

        print(
            f"Epoch {epoch:02d}/{cfg.epochs} | "
            f"train {tr:.4f} | val {vl:.4f} | ppl {ppl:.2f}"
        )

        if vl < best_val:
            best_val = vl
            torch.save(
                {
                    "model_state": model.state_dict(),
                    "itos": itos,
                    "stoi": stoi,
                    "config": cfg.__dict__,
                },
                cfg.save_path,
            )
            print(f"  → сохранено: {cfg.save_path}")

    print("\nОбучение завершено.")
    print(f"Лучший val_loss: {best_val:.4f}")


if __name__ == "__main__":
    main()

"""
generate.py — генерация текста обученной моделью.

Запуск:
    python generate.py --prompt "нейросет" --tokens 300
    python generate.py --prompt "" --temperature 0.7 --top-k 5
"""

import argparse
import torch
import torch.nn as nn
import torch.nn.functional as F


# --- должны совпадать с train.py ---
class CharLSTM(nn.Module):
    def __init__(self, vocab_size, emb_dim, hidden_dim, num_layers, dropout):
        super().__init__()
        self.emb = nn.Embedding(vocab_size, emb_dim)
        self.lstm = nn.LSTM(
            emb_dim,
            hidden_dim,
            num_layers,
            batch_first=True,
            dropout=dropout if num_layers > 1 else 0.0,
        )
        self.drop = nn.Dropout(dropout)
        self.fc = nn.Linear(hidden_dim, vocab_size)

    def forward(self, x, hidden=None):
        emb = self.emb(x)
        out, hidden = self.lstm(emb, hidden)
        out = self.drop(out)
        return self.fc(out), hidden


@torch.no_grad()
def generate(model, stoi, itos, device, start_str="",
             max_new_tokens=300, temperature=0.8, top_k=10,
             block_size=128):
    model.eval()

    if start_str:
        ids = [stoi.get(ch, stoi["<unk>"]) for ch in start_str.lower()]
    else:
        ids = [stoi["<bos>"]]

    x = torch.tensor(ids, dtype=torch.long, device=device).unsqueeze(0)
    out_ids = list(ids)

    for _ in range(max_new_tokens):
        x_cond = x[:, -block_size:]
        logits, _ = model(x_cond)
        logits = logits[:, -1, :] / temperature

        if top_k is not None:
            v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
            logits[logits < v[:, [-1]]] = -float("inf")

        probs = F.softmax(logits, dim=-1)
        next_id = torch.multinomial(probs, num_samples=1)

        if next_id.item() == stoi.get("<eos>", -1):
            break

        out_ids.append(next_id.item())
        x = torch.cat([x, next_id], dim=1)

    chars = []
    for i in out_ids:
        ch = itos[i]
        if ch in ("<pad>", "<bos>", "<eos>"):
            continue
        if ch == "<unk>":
            ch = "?"
        chars.append(ch)
    return "".join(chars)


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--model", type=str, default="best_char_lm.pt")
    p.add_argument("--prompt", type=str, default="")
    p.add_argument("--tokens", type=int, default=300)
    p.add_argument("--temperature", type=float, default=0.8)
    p.add_argument("--top-k", type=int, default=10)
    return p.parse_args()


def main():
    args = parse_args()
    device = "cuda" if torch.cuda.is_available() else "cpu"

    ckpt = torch.load(args.model, map_location=device)
    itos = ckpt["itos"]
    stoi = ckpt["stoi"]
    cfg = ckpt["config"]

    model = CharLSTM(
        vocab_size=len(itos),
        emb_dim=cfg["emb_dim"],
        hidden_dim=cfg["hidden_dim"],
        num_layers=cfg["num_layers"],
        dropout=cfg["dropout"],
    ).to(device)
    model.load_state_dict(ckpt["model_state"])

    print(f"Модель загружена: {args.model}")
    print(f"Промпт: {args.prompt!r}\n")

    text = generate(
        model, stoi, itos, device,
        start_str=args.prompt,
        max_new_tokens=args.tokens,
        temperature=args.temperature,
        top_k=args.top_k,
        block_size=cfg["block_size"],
    )
    print(text)


if __name__ == "__main__":
    main()

"""
download_wiki.py — скачивание и очистка русской Википедии.

Запуск:
    python download_wiki.py --output wiki_corpus.txt --max-mb 100
"""

import os
import re
import bz2
import argparse
import urllib.request
import xml.etree.ElementTree as ET


WIKI_URL = "https://dumps.wikimedia.org/ruwiki/latest/ruwiki-latest-pages-articles.xml.bz2"
DUMP_FILE = "ruwiki-latest-pages-articles.xml.bz2"


def download_dump(url: str = WIKI_URL):
    if os.path.exists(DUMP_FILE):
        print(f"Дамп уже есть: {DUMP_FILE}")
        return DUMP_FILE
    print(f"Скачиваю {url}")
    print("Это может занять несколько минут (файл ~3 ГБ)...")
    urllib.request.urlretrieve(url, DUMP_FILE)
    print(f"Скачано: {DUMP_FILE}")
    return DUMP_FILE


def clean_wiki_text(text: str) -> str:
    """Убирает вики-разметку, ссылки, шаблоны."""
    text = re.sub(r"<[^>]+>", " ", text)                  # html-теги
    text = re.sub(r"\{\{[^}]*\}\}", " ", text)            # {{шаблоны}}
    text = re.sub(r"\[\[([^\]|]+)\|([^\]]+)\]\]", r"\2", text)  # [[a|b]] -> b
    text = re.sub(r"\[\[([^\]]+)\]\]", r"\1", text)       # [[a]] -> a
    text = re.sub(r"\[https?://\S+\s+([^\]]+)\]", r"\1", text)  # [url text]
    text = re.sub(r"\[https?://\S+\]", " ", text)         # [url]
    text = re.sub(r"'{2,}", "", text)                     # '''bold'''
    text = re.sub(r"&nbsp;", " ", text)
    text = re.sub(r"&quot;", '"', text)
    text = re.sub(r"&amp;", "&", text)
    text = re.sub(r"&lt;", "<", text)
    text = re.sub(r"&gt;", ">", text)
    text = re.sub(r"\s+", " ", text)
    return text.strip()


def extract_text(dump_path: str, output: str, max_mb: int = 100):
    """Потоковый парсинг bz2-дампа. Останавливается при достижении max_mb."""
    max_bytes = max_mb * 1024 * 1024
    written = 0
    pages = 0

    print(f"Парсинг {dump_path} → {output} (лимит {max_mb} МБ)")

    with bz2.open(dump_path, "rt", encoding="utf-8") as f, \
         open(output, "w", encoding="utf-8") as out:

        in_page = False
        in_text = False
        buffer = []

        for event, elem in ET.iterparse(f, events=("start", "end")):
            tag = elem.tag.split("}")[-1]

            if event == "start":
                if tag == "page":
                    in_page = True
                    buffer = []
                elif tag == "text" and in_page:
                    in_text = True
                    buffer = [elem.text or ""]
            elif event == "end":
                if tag == "text" and in_text:
                    raw = "".join(buffer)
                    clean = clean_wiki_text(raw)
                    if len(clean) > 200:
                        out.write(clean + "\n\n")
                        written += len(clean.encode("utf-8"))
                        pages += 1
                    in_text = False
                    elem.clear()
                elif tag == "page":
                    in_page = False
                    elem.clear()
                    if pages % 1000 == 0:
                        print(f"  страниц: {pages}, {written/1024/1024:.1f} МБ")

            if written >= max_bytes:
                print(f"Достигнут лимит {max_mb} МБ. Останавливаюсь.")
                break

    print(f"Готово. Страниц: {pages}, размер: {written/1024/1024:.1f} МБ")


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--output", type=str, default="wiki_corpus.txt")
    p.add_argument("--max-mb", type=int, default=100)
    p.add_argument("--skip-download", action="store_true")
    return p.parse_args()


def main():
    args = parse_args()
    if not args.skip_download:
        download_dump()
    extract_text(DUMP_FILE, args.output, args.max_mb)


if __name__ == "__main__":
    main()

"""
train_transformer.py — обучение Transformer-языковой модели на русском.

Особенности:
- BPE-токенизация (SentencePiece)
- decoder-only Transformer (GPT-подобный)
- mixed precision (AMP) на GPU
- cosine schedule с warmup
- gradient accumulation

Запуск:
    python train_transformer.py --data wiki_corpus.txt --tokenizer bpe.model
"""

import os
import math
import time
import random
import argparse
from dataclasses import dataclass

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import Dataset, DataLoader
import sentencepiece as spm


# =====================================================================
# Конфигурация
# =====================================================================
@dataclass
class Config:
    data_path: str = "wiki_corpus.txt"
    tokenizer_path: str = "bpe.model"
    save_path: str = "best_transformer.pt"
    block_size: int = 256
    batch_size: int = 32
    grad_accum: int = 4
    n_layer: int = 6
    n_head: int = 6
    n_embd: int = 384
    dropout: float = 0.1
    lr: float = 3e-4
    min_lr: float = 3e-5
    warmup_steps: int = 500
    weight_decay: float = 0.1
    epochs: int = 5
    grad_clip: float = 1.0
    seed: int = 42
    val_ratio: float = 0.02
    use_amp: bool = True
    device: str = "cuda" if torch.cuda.is_available() else "cpu"
    log_every: int = 50
    eval_every: int = 500


# =====================================================================
# Dataset — читаем весь корпус в один тензор токенов
# =====================================================================
class TokenDataset(Dataset):
    def __init__(self, data: torch.Tensor, block_size: int):
        self.data = data
        self.block_size = block_size

    def __len__(self):
        return max(0, len(self.data) - self.block_size - 1)

    def __getitem__(self, idx):
        x = self.data[idx: idx + self.block_size]
        y = self.data[idx + 1: idx + self.block_size + 1]
        return x, y


# =====================================================================
# Transformer (GPT-подобный)
# =====================================================================
class CausalSelfAttention(nn.Module):
    def __init__(self, n_embd, n_head, dropout, block_size):
        super().__init__()
        assert n_embd % n_head == 0
        self.n_head = n_head
        self.n_embd = n_embd
        self.head_dim = n_embd // n_head

        self.qkv = nn.Linear(n_embd, 3 * n_embd, bias=False)
        self.proj = nn.Linear(n_embd, n_embd, bias=False)
        self.attn_drop = nn.Dropout(dropout)
        self.proj_drop = nn.Dropout(dropout)

        # маска причинности
        self.register_buffer(
            "mask",
            torch.tril(torch.ones(block_size, block_size)).view(
                1, 1, block_size, block_size
            ),
            persistent=False,
        )

    def forward(self, x):
        B, T, C = x.shape
        qkv = self.qkv(x)                       # (B, T, 3C)
        q, k, v = qkv.chunk(3, dim=-1)

        q = q.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        k = k.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        v = v.view(B, T, self.n_head, self.head_dim).transpose(1, 2)

        # scaled dot-product attention
        att = (q @ k.transpose(-2, -1)) / math.sqrt(self.head_dim)
        att = att.masked_fill(self.mask[:, :, :T, :T] == 0, float("-inf"))
        att = F.softmax(att, dim=-1)
        att = self.attn_drop(att)

        out = att @ v                            # (B, nh, T, hd)
        out = out.transpose(1, 2).contiguous().view(B, T, C)
        return self.proj_drop(self.proj(out))


class MLP(nn.Module):
    def __init__(self, n_embd, dropout):
        super().__init__()
        self.fc1 = nn.Linear(n_embd, 4 * n_embd)
        self.fc2 = nn.Linear(4 * n_embd, n_embd)
        self.drop = nn.Dropout(dropout)

    def forward(self, x):
        return self.drop(self.fc2(F.gelu(self.fc1(x))))


class Block(nn.Module):
    def __init__(self, n_embd, n_head, dropout, block_size):
        super().__init__()
        self.ln1 = nn.LayerNorm(n_embd)
        self.attn = CausalSelfAttention(n_embd, n_head, dropout, block_size)
        self.ln2 = nn.LayerNorm(n_embd)
        self.mlp = MLP(n_embd, dropout)

    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        x = x + self.mlp(self.ln2(x))
        return x


class GPT(nn.Module):
    def __init__(self, vocab_size, n_layer, n_head, n_embd, block_size, dropout):
        super().__init__()
        self.block_size = block_size
        self.tok_emb = nn.Embedding(vocab_size, n_embd)
        self.pos_emb = nn.Embedding(block_size, n_embd)
        self.drop = nn.Dropout(dropout)
        self.blocks = nn.Sequential(*[
            Block(n_embd, n_head, dropout, block_size) for _ in range(n_layer)
        ])
        self.ln_f = nn.LayerNorm(n_embd)
        self.head = nn.Linear(n_embd, vocab_size, bias=False)

        # weight tying
        self.head.weight = self.tok_emb.weight

        self.apply(self._init_weights)

    def _init_weights(self, m):
        if isinstance(m, nn.Linear):
            nn.init.normal_(m.weight, mean=0.0, std=0.02)
            if m.bias is not None:
                nn.init.zeros_(m.bias)
        elif isinstance(m, nn.Embedding):
            nn.init.normal_(m.weight, mean=0.0, std=0.02)

    def forward(self, x, targets=None):
        B, T = x.shape
        pos = torch.arange(T, device=x.device).unsqueeze(0)
        x = self.drop(self.tok_emb(x) + self.pos_emb(pos))
        x = self.blocks(x)
        x = self.ln_f(x)
        logits = self.head(x)

        loss = None
        if targets is not None:
            loss = F.cross_entropy(
                logits.view(-1, logits.size(-1)),
                targets.view(-1),
                ignore_index=0,  # <pad>
            )
        return logits, loss

    @torch.no_grad()
    def generate(self, idx, max_new_tokens, temperature=0.8, top_k=40):
        for _ in range(max_new_tokens):
            idx_cond = idx[:, -self.block_size:]
            logits, _ = self(idx_cond)
            logits = logits[:, -1, :] / temperature
            if top_k is not None:
                v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
                logits[logits < v[:, [-1]]] = -float("inf")
            probs = F.softmax(logits, dim=-1)
            next_id = torch.multinomial(probs, num_samples=1)
            idx = torch.cat([idx, next_id], dim=1)
        return idx

def get_lr(step, cfg, total_steps):
    if step < cfg.warmup_steps:
        return cfg.lr * step / max(1, cfg.warmup_steps)
    if step > total_steps:
        return cfg.min_lr
    ratio = (step - cfg.warmup_steps) / max(1, total_steps - cfg.warmup_steps)
    coeff = 0.5 * (1.0 + math.cos(math.pi * ratio))
    return cfg.min_lr + coeff * (cfg.lr - cfg.min_lr)


def encode_corpus(text, sp, eos_id):
    """Кодируем корпус целиком в список id (потоково по строкам)."""
    ids = []
    for line in text.split("\n"):
        line = line.strip()
        if not line:
            continue
        ids.extend(sp.encode(line, out_type=int))
        ids.append(eos_id)
    return ids

def main():
    args = parse_args()

    cfg = Config(
        data_path=args.data,
        tokenizer_path=args.tokenizer,
        save_path=args.save,
        epochs=args.epochs,
        batch_size=args.batch_size,
        block_size=args.block_size,
        grad_accum=args.grad_accum,
        lr=args.lr,
        n_layer=args.n_layer,
        n_head=args.n_head,
        n_embd=args.n_embd,
        dropout=args.dropout,
        use_amp=not args.no_amp,
    )

    random.seed(cfg.seed)
    torch.manual_seed(cfg.seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(cfg.seed)

    print(f"Устройство: {cfg.device}")
    print(f"AMP: {cfg.use_amp and cfg.device == 'cuda'}")

    # 1. Токенизатор
    if not os.path.exists(cfg.tokenizer_path):
        raise FileNotFoundError(
            f"Не найден {cfg.tokenizer_path}. Сначала запустите train_bpe.py"
        )
    sp = spm.SentencePieceProcessor(model_file=cfg.tokenizer_path)
    vocab_size = sp.get_piece_size()
    eos_id = sp.eos_id()
    print(f"Словарь BPE: {vocab_size}")

    # 2. Корпус
    if not os.path.exists(cfg.data_path):
        raise FileNotFoundError(f"Не найден {cfg.data_path}")
    with open(cfg.data_path, "r", encoding="utf-8") as f:
        text = f.read()
    print(f"Корпус: {len(text):,} символов")

    ids = encode_corpus(text, sp, eos_id)
    data = torch.tensor(ids, dtype=torch.long)
    print(f"Токенов: {len(data):,}")

    if len(data) < cfg.block_size * 10:
        raise ValueError("Корпус слишком маленький для Transformer.")

    # 3. Split
    n = len(data)
    n_val = max(cfg.block_size * 2, int(n * cfg.val_ratio))
    n_train = n - n_val
    train_data = data[:n_train]
    val_data = data[n_train:]
    print(f"Train: {len(train_data):,} | Val: {len(val_data):,}")

    train_loader = DataLoader(
        TokenDataset(train_data, cfg.block_size),
        batch_size=cfg.batch_size,
        shuffle=True,
        drop_last=True,
        num_workers=2,
        pin_memory=True,
    )
    val_loader = DataLoader(
        TokenDataset(val_data, cfg.block_size),
        batch_size=cfg.batch_size,
        shuffle=False,
        drop_last=False,
        num_workers=2,
        pin_memory=True,
    )

    # 4. Модель
    model = GPT(
        vocab_size=vocab_size,
        n_layer=cfg.n_layer,
        n_head=cfg.n_head,
        n_embd=cfg.n_embd,
        block_size=cfg.block_size,
        dropout=cfg.dropout,
    ).to(cfg.device)

    n_params = sum(p.numel() for p in model.parameters())
    print(f"Параметров: {n_params:,} ({n_params/1e6:.2f}M)")

    # оптимизатор: без weight decay для LayerNorm и bias
    decay, no_decay = [], []
    for name, p in model.named_parameters():
        if not p.requires_grad:
            continue
        if p.dim() >= 2:
            decay.append(p)
        else:
            no_decay.append(p)

    optimizer = torch.optim.AdamW(
        [
            {"params": decay, "weight_decay": cfg.weight_decay},
            {"params": no_decay, "weight_decay": 0.0},
        ],
        lr=cfg.lr,
        betas=(0.9, 0.95),
    )

    # AMP
    use_amp = cfg.use_amp and cfg.device == "cuda"
    scaler = torch.cuda.amp.GradScaler(enabled=use_amp)

    # шаги
    steps_per_epoch = len(train_loader) // cfg.grad_accum
    total_steps = steps_per_epoch * cfg.epochs
    print(f"Шагов на эпоху: {steps_per_epoch}, всего: {total_steps}")

    # 5. Обучение
    best_val = float("inf")
    global_step = 0
    t0 = time.time()

    for epoch in range(1, cfg.epochs + 1):
        model.train()
        running = 0.0
        optimizer.zero_grad(set_to_none=True)

        for it, (x, y) in enumerate(train_loader):
            x = x.to(cfg.device, non_blocking=True)
            y = y.to(cfg.device, non_blocking=True)

            with torch.cuda.amp.autocast(enabled=use_amp):
                _, loss = model(x, y)
                loss = loss / cfg.grad_accum

            scaler.scale(loss).backward()
            running += loss.item()

            if (it + 1) % cfg.grad_accum == 0:
                lr = get_lr(global_step, cfg, total_steps)
                for g in optimizer.param_groups:
                    g["lr"] = lr

                scaler.unscale_(optimizer)
                torch.nn.utils.clip_grad_norm_(model.parameters(), cfg.grad_clip)

                scaler.step(optimizer)
                scaler.update()
                optimizer.zero_grad(set_to_none=True)

                global_step += 1

                if global_step % cfg.log_every == 0:
                    dt = time.time() - t0
                    print(
                        f"ep {epoch} | step {global_step}/{total_steps} | "
                        f"loss {running / cfg.log_every:.4f} | "
                        f"lr {lr:.2e} | {dt:.0f}s"
                    )
                    running = 0.0

                if global_step % cfg.eval_every == 0:
                    vl = evaluate(model, val_loader, cfg, use_amp)
                    print(f"  >> val_loss {vl:.4f} | ppl {math.exp(vl):.2f}")
                    if vl < best_val:
                        best_val = vl
                        torch.save(
                            {
                                "model_state": model.state_dict(),
                                "config": cfg.__dict__,
                                "vocab_size": vocab_size,
                                "tokenizer_path": cfg.tokenizer_path,
                            },
                            cfg.save_path,
                        )
                        print(f"  >> сохранено: {cfg.save_path}")
                    model.train()

        # конец эпохи — финальная валидация
        vl = evaluate(model, val_loader, cfg, use_amp)
        print(f"== Эпоха {epoch} завершена | val_loss {vl:.4f} | ppl {math.exp(vl):.2f}")
        if vl < best_val:
            best_val = vl
            torch.save(
                {
                    "model_state": model.state_dict(),
                    "config": cfg.__dict__,
                    "vocab_size": vocab_size,
                    "tokenizer_path": cfg.tokenizer_path,
                },
                cfg.save_path,
            )
            print(f"  >> сохранено: {cfg.save_path}")

    print(f"\nОбучение завершено за {(time.time()-t0)/60:.1f} мин")
    print(f"Лучший val_loss: {best_val:.4f}")


@torch.no_grad()
def evaluate(model, loader, cfg, use_amp):
    model.eval()
    total = 0.0
    count = 0
    for x, y in loader:
        x = x.to(cfg.device, non_blocking=True)
        y = y.to(cfg.device, non_blocking=True)
        with torch.cuda.amp.autocast(enabled=use_amp):
            _, loss = model(x, y)
        total += loss.item()
        count += 1
        if count >= 50:  # ограничимся 50 батчами для скорости
            break
    return total / max(1, count)

    main()

import torch
import torch.nn as nn
import torch.nn.functional as F
import math
import sentencepiece as spm


# --- дублируем архитектуру, чтобы загрузить веса ---
class CausalSelfAttention(nn.Module):
    def __init__(self, n_embd, n_head, dropout, block_size):
        super().__init__()
        self.n_head = n_head
        self.head_dim = n_embd // n_head
        self.qkv = nn.Linear(n_embd, 3 * n_embd, bias=False)
        self.proj = nn.Linear(n_embd, n_embd, bias=False)
        self.attn_drop = nn.Dropout(dropout)
        self.proj_drop = nn.Dropout(dropout)
        self.register_buffer(
            "mask",
            torch.tril(torch.ones(block_size, block_size)).view(
                1, 1, block_size, block_size
            ),
            persistent=False,
        )

    def forward(self, x):
        B, T, C = x.shape
        q, k, v = self.qkv(x).chunk(3, dim=-1)
        q = q.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        k = k.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        v = v.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        att = (q @ k.transpose(-2, -1)) / math.sqrt(self.head_dim)
        att = att.masked_fill(self.mask[:, :, :T, :T] == 0, float("-inf"))
        att = F.softmax(att, dim=-1)
        att = self.attn_drop(att)
        out = att @ v
        out = out.transpose(1, 2).contiguous().view(B, T, C)
        return self.proj_drop(self.proj(out))


class MLP(nn.Module):
    def __init__(self, n_embd, dropout):
        super().__init__()
        self.fc1 = nn.Linear(n_embd, 4 * n_embd)
        self.fc2 = nn.Linear(4 * n_embd, n_embd)
        self.drop = nn.Dropout(dropout)

    def forward(self, x):
        return self.drop(self.fc2(F.gelu(self.fc1(x))))


class Block(nn.Module):
    def __init__(self, n_embd, n_head, dropout, block_size):
        super().__init__()
        self.ln1 = nn.LayerNorm(n_embd)
        self.attn = CausalSelfAttention(n_embd, n_head, dropout, block_size)
        self.ln2 = nn.LayerNorm(n_embd)
        self.mlp = MLP(n_embd, dropout)

    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        x = x + self.mlp(self.ln2(x))
        return x


class GPT(nn.Module):
    def __init__(self, vocab_size, n_layer, n_head, n_embd, block_size, dropout):
        super().__init__()
        self.block_size = block_size
        self.tok_emb = nn.Embedding(vocab_size, n_embd)
        self.pos_emb = nn.Embedding(block_size, n_embd)
        self.drop = nn.Dropout(dropout)
        self.blocks = nn.Sequential(*[
            Block(n_embd, n_head, dropout, block_size) for _ in range(n_layer)
        ])
        self.ln_f = nn.LayerNorm(n_embd)
        self.head = nn.Linear(n_embd, vocab_size, bias=False)
        self.head.weight = self.tok_emb.weight

    @torch.no_grad()
    def generate(self, idx, max_new_tokens, temperature=0.8, top_k=40,
                 eos_id=None, repetition_penalty=1.0):
        for _ in range(max_new_tokens):
            idx_cond = idx[:, -self.block_size:]
            logits, _ = self(idx_cond)
            logits = logits[:, -1, :]

            if repetition_penalty != 1.0:
                for i in range(idx.size(0)):
                    seen = set(idx[i].tolist())
                    for tok in seen:
                        if logits[i, tok] > 0:
                            logits[i, tok] /= repetition_penalty
                        else:
                            logits[i, tok] *= repetition_penalty

            logits = logits / temperature
            if top_k is not None:
                v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
                logits[logits < v[:, [-1]]] = -float("inf")

            probs = F.softmax(logits, dim=-1)
            next_id = torch.multinomial(probs, num_samples=1)
            idx = torch.cat([idx, next_id], dim=1)

            if eos_id is not None and (next_id == eos_id).all():
                break
        return idx

    p.add_argument("--model", type=str, default="best_transformer.pt")
    p.add_argument("--tokenizer", type=str, default=None)

    device = "cuda" if torch.cuda.is_available() else "cpu"

    ckpt = torch.load(args.model, map_location=device)

    model.eval()

    print(f"Промпт: {args.prompt!r}")
    print(f"Параметры: temp={args.temperature}, top_k={args.top_k}, "
          f"rep_pen={args.repetition_penalty}\n")

    for s in range(args.num_samples):
        ids = sp.encode(args.prompt, out_type=int)
        idx = torch.tensor(ids, dtype=torch.long, device=device).unsqueeze(0)

            repetition_penalty=args.repetition_penalty,

        text = sp.decode(out[0].tolist())
        print(f"--- Пример {s+1} ---")
        print(text)
        print()

"""
model_gpt.py — GPT-подобная модель с поддержкой KV-cache.

Веса совместимы с train_transformer.py (те же имена слоёв).
"""

import math
import torch
import torch.nn as nn
import torch.nn.functional as F


class CausalSelfAttention(nn.Module):
    def __init__(self, n_embd, n_head, dropout, block_size):
        super().__init__()
        assert n_embd % n_head == 0
        self.n_head = n_head
        self.head_dim = n_embd // n_head
        self.qkv = nn.Linear(n_embd, 3 * n_embd, bias=False)
        self.proj = nn.Linear(n_embd, n_embd, bias=False)
        self.attn_drop = nn.Dropout(dropout)
        self.proj_drop = nn.Dropout(dropout)
        self.block_size = block_size

        self.register_buffer(
            "mask",
            torch.tril(torch.ones(block_size, block_size)).view(
                1, 1, block_size, block_size
            ),
            persistent=False,
        )

    def forward(self, x, kv_cache=None, use_cache=False):
        B, T, C = x.shape
        qkv = self.qkv(x)
        q, k, v = qkv.chunk(3, dim=-1)

        q = q.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        k = k.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        v = v.view(B, T, self.n_head, self.head_dim).transpose(1, 2)

        # KV-cache: приклеиваем прошлые K, V
        new_cache = None
        if use_cache:
            if kv_cache is not None:
                past_k, past_v = kv_cache
                k = torch.cat([past_k, k], dim=2)
                v = torch.cat([past_v, v], dim=2)
            new_cache = (k, v)

        T_q = q.size(2)          # длина запроса
        T_k = k.size(2)          # длина ключей (с кэшем больше)

        att = (q @ k.transpose(-2, -1)) / math.sqrt(self.head_dim)

        # Маска причинности:
        # - при обучении T_q == T_k -> используем квадратную маску
        # - при генерации с кэшем T_q == 1 -> маска не нужна
        if T_q > 1:
            mask = self.mask[:, :, :T_q, :T_k]
            att = att.masked_fill(mask == 0, float("-inf"))

        att = F.softmax(att, dim=-1)
        att = self.attn_drop(att)

        out = att @ v
        out = out.transpose(1, 2).contiguous().view(B, T, C)
        out = self.proj_drop(self.proj(out))

        if use_cache:
            return out, new_cache
        return out, None


class MLP(nn.Module):
    def __init__(self, n_embd, dropout):
        super().__init__()
        self.fc1 = nn.Linear(n_embd, 4 * n_embd)
        self.fc2 = nn.Linear(4 * n_embd, n_embd)
        self.drop = nn.Dropout(dropout)

    def forward(self, x):
        return self.drop(self.fc2(F.gelu(self.fc1(x))))


class Block(nn.Module):
    def __init__(self, n_embd, n_head, dropout, block_size):
        super().__init__()
        self.ln1 = nn.LayerNorm(n_embd)
        self.attn = CausalSelfAttention(n_embd, n_head, dropout, block_size)
        self.ln2 = nn.LayerNorm(n_embd)
        self.mlp = MLP(n_embd, dropout)

    def forward(self, x, kv_cache=None, use_cache=False):
        attn_out, new_cache = self.attn(self.ln1(x), kv_cache, use_cache)
        x = x + attn_out
        x = x + self.mlp(self.ln2(x))
        return x, new_cache


class GPT(nn.Module):
    def __init__(self, vocab_size, n_layer, n_head, n_embd, block_size, dropout):
        super().__init__()
        self.block_size = block_size
        self.tok_emb = nn.Embedding(vocab_size, n_embd)
        self.pos_emb = nn.Embedding(block_size, n_embd)
        self.drop = nn.Dropout(dropout)
        self.blocks = nn.ModuleList([
            Block(n_embd, n_head, dropout, block_size) for _ in range(n_layer)
        ])
        self.ln_f = nn.LayerNorm(n_embd)
        self.head = nn.Linear(n_embd, vocab_size, bias=False)
        self.head.weight = self.tok_emb.weight  # weight tying
        self.apply(self._init_weights)

    def _init_weights(self, m):
        if isinstance(m, nn.Linear):
            nn.init.normal_(m.weight, mean=0.0, std=0.02)
            if m.bias is not None:
                nn.init.zeros_(m.bias)
        elif isinstance(m, nn.Embedding):
            nn.init.normal_(m.weight, mean=0.0, std=0.02)

    def forward(self, x, targets=None, kv_caches=None, use_cache=False,
                pos_offset=0):
        B, T = x.shape
        pos = torch.arange(pos_offset, pos_offset + T, device=x.device).unsqueeze(0)
        x = self.drop(self.tok_emb(x) + self.pos_emb(pos))

        new_caches = [] if use_cache else None
        for i, block in enumerate(self.blocks):
            past = kv_caches[i] if kv_caches is not None else None
            x, c = block(x, past, use_cache)
            if use_cache:
                new_caches.append(c)

        x = self.ln_f(x)
        logits = self.head(x)

        loss = None
        if targets is not None:
            loss = F.cross_entropy(
                logits.view(-1, logits.size(-1)),
                targets.view(-1),
                ignore_index=0,
            )
        return logits, loss, new_caches

    # ---------------- Генерация без кэша (простая, для отладки) --------
    @torch.no_grad()
    def generate_naive(self, idx, max_new_tokens, temperature=0.8, top_k=40,
                       eos_id=None, repetition_penalty=1.0):
        for _ in range(max_new_tokens):
            idx_cond = idx[:, -self.block_size:]
            logits, _, _ = self(idx_cond)
            logits = logits[:, -1, :]
            logits = apply_penalties(logits, idx, temperature,
                                     top_k, repetition_penalty)
            probs = F.softmax(logits, dim=-1)
            nxt = torch.multinomial(probs, num_samples=1)
            idx = torch.cat([idx, nxt], dim=1)
            if eos_id is not None and (nxt == eos_id).all():
                break
        return idx

    # ---------------- Генерация с KV-cache (быстрая) -------------------
    @torch.no_grad()
    def generate(self, idx, max_new_tokens, temperature=0.8, top_k=40,
                 eos_id=None, repetition_penalty=1.0):
        self.eval()

        # 1) Прогоняем промпт, получаем кэш
        idx_cond = idx[:, -self.block_size:]
        logits, _, caches = self(idx_cond, use_cache=True)
        logits = logits[:, -1, :]
        pos = idx_cond.size(1)

        for _ in range(max_new_tokens):
            logits = apply_penalties(logits, idx, temperature,
                                     top_k, repetition_penalty)
            probs = F.softmax(logits, dim=-1)
            nxt = torch.multinomial(probs, num_samples=1)  # (B, 1)
            idx = torch.cat([idx, nxt], dim=1)

            if eos_id is not None and (nxt == eos_id).all():
                break

            # обрезаем кэш, чтобы не выйти за block_size
            if pos >= self.block_size:
                # сдвигаем кэш: отрезаем самое старое
                caches = [
                    (k[:, :, -self.block_size + 1:, :],
                     v[:, :, -self.block_size + 1:, :])
                    for k, v in caches
                ]
                pos = self.block_size - 1

            # 2) Прогоняем один токен через модель, обновляем кэш
            logits, _, caches = self(
                nxt, kv_caches=caches, use_cache=True, pos_offset=pos
            )
            logits = logits[:, -1, :]
            pos += 1

        return idx


def apply_penalties(logits, idx, temperature, top_k, repetition_penalty):
    """Применяет repetition penalty, temperature, top-k."""
    if repetition_penalty != 1.0:
        for i in range(idx.size(0)):
            seen = set(idx[i].tolist())
            for tok in seen:
                if logits[i, tok] > 0:
                    logits[i, tok] /= repetition_penalty
                else:
                    logits[i, tok] *= repetition_penalty

    logits = logits / temperature
    if top_k is not None:
        v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
        logits[logits < v[:, [-1]]] = -float("inf")
    return logits


def load_model(ckpt_path, device="cpu", dropout=0.0):
    """Загружает модель из чекпоинта train_transformer.py."""
    ckpt = torch.load(ckpt_path, map_location=device)
    cfg = ckpt["config"]
    model = GPT(
        vocab_size=ckpt["vocab_size"],
        n_layer=cfg["n_layer"],
        n_head=cfg["n_head"],
        n_embd=cfg["n_embd"],
        block_size=cfg["block_size"],
        dropout=dropout,
    ).to(device)
    model.load_state_dict(ckpt["model_state"])
    model.eval()
    return model, ckpt

"""
generate_fast.py — генерация с KV-cache.

Запуск:
    python generate_fast.py --prompt "Нейросеть — это" --tokens 200
    python generate_fast.py --prompt "В лесу" --benchmark
"""

import time
import argparse
import torch
import sentencepiece as spm

from model_gpt import load_model


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--model", type=str, default="best_transformer.pt")
    p.add_argument("--tokenizer", type=str, default=None)
    p.add_argument("--prompt", type=str, default="Нейросеть — это")
    p.add_argument("--tokens", type=int, default=200)
    p.add_argument("--temperature", type=float, default=0.85)
    p.add_argument("--top-k", type=int, default=40)
    p.add_argument("--repetition-penalty", type=float, default=1.1)
    p.add_argument("--num-samples", type=int, default=1)
    p.add_argument("--benchmark", action="store_true",
                   help="сравнить скорость с/без KV-cache")
    return p.parse_args()


def main():
    args = parse_args()
    device = "cuda" if torch.cuda.is_available() else "cpu"

    model, ckpt = load_model(args.model, device=device)
    tok_path = args.tokenizer or ckpt.get("tokenizer_path", "bpe.model")
    sp = spm.SentencePieceProcessor(model_file=tok_path)

    print(f"Модель: {args.model} | устройство: {device}")
    print(f"Промпт: {args.prompt!r}\n")

    if args.benchmark:
        run_benchmark(model, sp, args, device)
        return

    for s in range(args.num_samples):
        ids = sp.encode(args.prompt, out_type=int)
        idx = torch.tensor(ids, dtype=torch.long, device=device).unsqueeze(0)

        t0 = time.time()
        out = model.generate(
            idx,
            max_new_tokens=args.tokens,
            temperature=args.temperature,
            top_k=args.top_k,
            eos_id=sp.eos_id(),
            repetition_penalty=args.repetition_penalty,
        )
        dt = time.time() - t0
        n_new = out.size(1) - idx.size(1)

        text = sp.decode(out[0].tolist())
        print(f"--- Пример {s+1} ({n_new} ток. за {dt:.2f}с, "
              f"{n_new/max(dt,1e-6):.1f} ток/с) ---")
        print(text)
        print()


@torch.no_grad()
def run_benchmark(model, sp, args, device):
    """Сравнивает наивную генерацию и KV-cache."""
    prompt_ids = sp.encode(args.prompt, out_type=int)
    idx = torch.tensor(prompt_ids, dtype=torch.long, device=device).unsqueeze(0)

    n = args.tokens

    # прогреем
    _ = model.generate(idx.clone(), max_new_tokens=5,
                       temperature=args.temperature, top_k=args.top_k,
                       eos_id=sp.eos_id())

    print("=== KV-cache ===")
    t0 = time.time()
    _ = model.generate(idx.clone(), max_new_tokens=n,
                       temperature=args.temperature, top_k=args.top_k,
                       eos_id=sp.eos_id())
    t_cache = time.time() - t0
    print(f"{n} токенов за {t_cache:.2f}с -> {n/t_cache:.1f} ток/с")

    print("\n=== Naive (без кэша) ===")
    t0 = time.time()
    _ = model.generate_naive(idx.clone(), max_new_tokens=n,
                             temperature=args.temperature, top_k=args.top_k,
                             eos_id=sp.eos_id())
    t_naive = time.time() - t0
    print(f"{n} токенов за {t_naive:.2f}с -> {n/t_naive:.1f} ток/с")

    print(f"\nУскорение: {t_naive / t_cache:.1f}x")


if __name__ == "__main__":
    main()

"""
app.py — веб-демо для обученной русской модели.

Запуск:
    python app.py
    python app.py --model best_transformer.pt --share
"""

import argparse
import torch
import gradio as gr
import sentencepiece as spm

from model_gpt import load_model


# -------------------- Глобальные переменные --------------------
MODEL = None
SP = None
DEVICE = "cuda" if torch.cuda.is_available() else "cpu"
CFG = {}


def init(model_path, tok_path=None):
    global MODEL, SP, CFG
    MODEL, ckpt = load_model(model_path, device=DEVICE)
    CFG = ckpt["config"]
    tok_path = tok_path or ckpt.get("tokenizer_path", "bpe.model")
    SP = spm.SentencePieceProcessor(model_file=tok_path)
    print(f"Модель загружена: {model_path} на {DEVICE}")
    print(f"vocab={ckpt['vocab_size']} | layers={CFG['n_layer']} | "
          f"embd={CFG['n_embd']}")


def generate(prompt, max_tokens, temperature, top_k, rep_pen, seed):
    if MODEL is None:
        return "Модель не загружена."

    if seed is not None and seed >= 0:
        torch.manual_seed(seed)

    if prompt.strip():
        ids = SP.encode(prompt, out_type=int)
    else:
        ids = [SP.bos_id()]

    idx = torch.tensor(ids, dtype=torch.long, device=DEVICE).unsqueeze(0)

    with torch.no_grad():
        out = MODEL.generate(
            idx,
            max_new_tokens=int(max_tokens),
            temperature=float(temperature),
            top_k=int(top_k),
            eos_id=SP.eos_id(),
            repetition_penalty=float(rep_pen),
        )

    text = SP.decode(out[0].tolist())
    return text


def build_ui():
    with gr.Blocks(title="Русская языковая модель", theme=gr.themes.Soft()) as demo:
        gr.Markdown(

        )

        with gr.Row():
            with gr.Column(scale=2):
                prompt = gr.Textbox(
                    label="Промпт",
                    value="Нейросеть — это",
                    lines=3,
                    placeholder="Введите начало текста...",
                )
                with gr.Row():
                    max_tokens = gr.Slider(20, 500, value=150, step=10,
                                           label="Макс. токенов")
                    temperature = gr.Slider(0.1, 1.5, value=0.85, step=0.05,
                                            label="Temperature")
                with gr.Row():
                    top_k = gr.Slider(1, 100, value=40, step=1, label="Top-K")
                    rep_pen = gr.Slider(1.0, 1.5, value=1.1, step=0.05,
                                        label="Repetition penalty")
                seed = gr.Number(value=-1, label="Seed (-1 = случайно)",
                                 precision=0)

                with gr.Row():
                    btn = gr.Button("Сгенерировать", variant="primary")
                    clear = gr.Button("Очистить")

            with gr.Column(scale=3):
                output = gr.Textbox(label="Результат", lines=20)

        gr.Examples(
            examples=[
                ["Нейросеть — это", 150, 0.85, 40, 1.1, -1],
                ["В лесу родилась", 150, 0.8, 40, 1.1, -1],
                ["Жил-был на свете", 200, 0.9, 50, 1.15, 42],
                ["Русский алфавит состоит из", 100, 0.7, 30, 1.1, -1],
            ],
            inputs=[prompt, max_tokens, temperature, top_k, rep_pen, seed],
        )

        btn.click(
            generate,
            inputs=[prompt, max_tokens, temperature, top_k, rep_pen, seed],
            outputs=output,
        )
        prompt.submit(
            generate,
            inputs=[prompt, max_tokens, temperature, top_k, rep_pen, seed],
            outputs=output,
        )
        clear.click(lambda: ("", ""), outputs=[prompt, output])

    return demo


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--model", type=str, default="best_transformer.pt")
    p.add_argument("--tokenizer", type=str, default=None)

"""
finetune_sft.py — Supervised Fine-Tuning (instruction tuning).

Запуск:
    python finetune_sft.py --base best_transformer.pt --data instructions.jsonl
"""

import os
import json
import math
import time
import random
import argparse
from dataclasses import dataclass

import torch
import torch.nn.functional as F
from torch.utils.data import Dataset, DataLoader
import sentencepiece as spm

from model_gpt import GPT, load_model


# =====================================================================
# Промпт-шаблон
# =====================================================================
PROMPT_TEMPLATE = "### Инструкция:\n{instruction}\n\n### Ответ:\n{response}<eos>"
TRAIN_TEMPLATE  = "### Инструкция:\n{instruction}\n\n### Ответ:\n{response}"


# =====================================================================
# Dataset
# =====================================================================
class SFTDataset(Dataset):
    """
    Возвращает (input_ids, labels).
    labels = -100 на токенах промпта, чтобы лосс считался только по ответу.
    """
    def __init__(self, samples, sp, block_size):
        self.samples = samples
        self.sp = sp
        self.block_size = block_size

    def __len__(self):
        return len(self.samples)

    def __getitem__(self, idx):
        s = self.samples[idx]
        prompt_text = PROMPT_TEMPLATE.format(
            instruction=s["instruction"], response=""
        ).replace("<eos>", "")
        full_text = PROMPT_TEMPLATE.format(
            instruction=s["instruction"], response=s["response"]
        )

        prompt_ids = self.sp.encode(prompt_text, out_type=int)
        full_ids = self.sp.encode(full_text, out_type=int)

        # обрезаем до block_size + 1
        if len(full_ids) > self.block_size + 1:
            full_ids = full_ids[: self.block_size + 1]

        x = full_ids[:-1]
        y = full_ids[1:]

        # маскируем промпт в labels
        prompt_len = len(prompt_ids)
        labels = [-100] * len(y)
        for i in range(prompt_len - 1, len(y)):
            labels[i] = y[i]

        # паддинг
        pad_len = self.block_size - len(x)
        if pad_len > 0:
            x = x + [0] * pad_len
            labels = labels + [-100] * pad_len

        return (
            torch.tensor(x, dtype=torch.long),
            torch.tensor(labels, dtype=torch.long),
        )


# =====================================================================
# Обучение
# =====================================================================
def train_epoch(model, loader, optimizer, scaler, cfg, use_amp):
    model.train()
    total, count = 0.0, 0
    for x, y in loader:
        x = x.to(cfg.device)
        y = y.to(cfg.device)

        with torch.cuda.amp.autocast(enabled=use_amp):
            _, loss, _ = model(x, targets=y)
            loss = loss / cfg.grad_accum

        scaler.scale(loss).backward()

        if (count + 1) % cfg.grad_accum == 0:
            scaler.unscale_(optimizer)
            torch.nn.utils.clip_grad_norm_(model.parameters(), cfg.grad_clip)
            scaler.step(optimizer)
            scaler.update()
            optimizer.zero_grad(set_to_none=True)

        total += loss.item() * cfg.grad_accum
        count += 1

    return total / max(1, count)


@torch.no_grad()
def evaluate(model, loader, cfg):
    model.eval()
    total, count = 0.0, 0
    for x, y in loader:
        x = x.to(cfg.device)
        y = y.to(cfg.device)
        _, loss, _ = model(x, targets=y)
        total += loss.item()
        count += 1
    return total / max(1, count)


# =====================================================================
# Main
# =====================================================================
@dataclass
class SFTConfig:
    base_model: str = "best_transformer.pt"
    data_path: str = "instructions.jsonl"
    tokenizer_path: str = "bpe.model"
    save_path: str = "sft_model.pt"
    block_size: int = 384
    batch_size: int = 8
    grad_accum: int = 4
    lr: float = 5e-5
    weight_decay: float = 0.01
    epochs: int = 3
    grad_clip: float = 1.0
    val_ratio: float = 0.05
    seed: int = 42
    device: str = "cuda" if torch.cuda.is_available() else "cpu"
    log_every: int = 20


def load_jsonl(path):
    samples = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            obj = json.loads(line)
            if "instruction" in obj and "response" in obj:
                samples.append(obj)
    return samples


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--base", type=str, default="best_transformer.pt")
    p.add_argument("--data", type=str, default="instructions.jsonl")
    p.add_argument("--tokenizer", type=str, default="bpe.model")
    p.add_argument("--save", type=str, default="sft_model.pt")
    p.add_argument("--epochs", type=int, default=3)
    p.add_argument("--batch-size", type=int, default=8)
    p.add_argument("--grad-accum", type=int, default=4)
    p.add_argument("--lr", type=float, default=5e-5)
    p.add_argument("--block-size", type=int, default=384)
    return p.parse_args()


def main():
    args = parse_args()
    cfg = SFTConfig(
        base_model=args.base,
        data_path=args.data,
        tokenizer_path=args.tokenizer,
        save_path=args.save,
        epochs=args.epochs,
        batch_size=args.batch_size,
        grad_accum=args.grad_accum,
        lr=args.lr,
        block_size=args.block_size,
    )

    random.seed(cfg.seed)
    torch.manual_seed(cfg.seed)

    print(f"Устройство: {cfg.device}")

    # 1. Модель и токенизатор
    model, ckpt = load_model(cfg.base_model, device=cfg.device)
    # перезаписываем block_size для SFT (может быть больше)
    if cfg.block_size > model.block_size:
        print(f"Расширяю block_size: {model.block_size} -> {cfg.block_size}")
        model.block_size = cfg.block_size
        for block in model.blocks:
            block.attn.block_size = cfg.block_size
            block.attn.mask = torch.tril(
                torch.ones(cfg.block_size, cfg.block_size)
            ).view(1, 1, cfg.block_size, cfg.block_size).to(cfg.device)
        # расширяем позиционные эмбеддинги
        old_pos = model.pos_emb.weight.data
        new_pos = torch.zeros(cfg.block_size, old_pos.size(1), device=cfg.device)
        new_pos[: old_pos.size(0)] = old_pos
        new_pos[old_pos.size(0):] = old_pos[-1:].repeat(
            cfg.block_size - old_pos.size(0), 1
        )
        model.pos_emb = torch.nn.Embedding(cfg.block_size, old_pos.size(1)).to(cfg.device)
        model.pos_emb.weight.data = new_pos

    tok_path = cfg.tokenizer_path or ckpt.get("tokenizer_path", "bpe.model")
    sp = spm.SentencePieceProcessor(model_file=tok_path)
    print(f"Словарь: {sp.get_piece_size()}")

    # 2. Данные
    if not os.path.exists(cfg.data_path):
        raise FileNotFoundError(f"Не найден {cfg.data_path}")

    samples = load_jsonl(cfg.data_path)
    random.shuffle(samples)
    print(f"Примеров: {len(samples)}")

    n_val = max(1, int(len(samples) * cfg.val_ratio))
    val_samples = samples[:n_val]
    train_samples = samples[n_val:]

    train_ds = SFTDataset(train_samples, sp, cfg.block_size)
    val_ds = SFTDataset(val_samples, sp, cfg.block_size)

    train_loader = DataLoader(train_ds, batch_size=cfg.batch_size,
                              shuffle=True, drop_last=True)
    val_loader = DataLoader(val_ds, batch_size=cfg.batch_size, shuffle=False)

    # 3. Оптимизатор
    decay, no_decay = [], []
    for n, p in model.named_parameters():
        if not p.requires_grad:
            continue
        (decay if p.dim() >= 2 else no_decay).append(p)

    optimizer = torch.optim.AdamW(
        [
            {"params": decay, "weight_decay": cfg.weight_decay},
            {"params": no_decay, "weight_decay": 0.0},
        ],
        lr=cfg.lr,
        betas=(0.9, 0.95),
    )

    use_amp = cfg.device == "cuda"
    scaler = torch.cuda.amp.GradScaler(enabled=use_amp)

    total_steps = len(train_loader) * cfg.epochs // cfg.grad_accum
    print(f"Шагов: {total_steps}")

    # 4. Обучение
    best_val = float("inf")
    t0 = time.time()
    step = 0

    for epoch in range(1, cfg.epochs + 1):
        tr_loss = train_epoch(model, train_loader, optimizer, scaler, cfg, use_amp)
        vl_loss = evaluate(model, val_loader, cfg)
        print(
            f"Epoch {epoch}/{cfg.epochs} | "
            f"train {tr_loss:.4f} | val {vl_loss:.4f} | "
            f"ppl {math.exp(vl_loss):.2f} | {time.time()-t0:.0f}s"
        )

        if vl_loss < best_val:
            best_val = vl_loss
            torch.save(
                {
                    "model_state": model.state_dict(),
                    "config": {**ckpt["config"], "block_size": cfg.block_size},
                    "vocab_size": ckpt["vocab_size"],
                    "tokenizer_path": tok_path,
                    "sft": True,
                },
                cfg.save_path,
            )
            print(f"  -> сохранено: {cfg.save_path}")

    print(f"\nГотово. Лучший val_loss: {best_val:.4f}")


if __name__ == "__main__":
    main()

class TinyGRU(nn.Module):
    def __init__(self, vocab_size, emb_dim, hidden_dim, num_layers, dropout):
        super().__init__()
        self.emb = nn.Embedding(vocab_size, emb_dim)
        self.gru = nn.GRU(emb_dim, hidden_dim, num_layers, batch_first=True, dropout=dropout if num_layers>1 else 0)
        self.fc = nn.Linear(hidden_dim, vocab_size)
    def forward(self, x, h=None):
        e = self.emb(x)
        o, h = self.gru(e, h)
        return self.fc(o), h

def get_batch(data, block_size, batch_size):
    ix = torch.randint(0, len(data) - block_size - 1, (batch_size,))
    x = torch.stack([data[i:i+block_size] for i in ix])
    y = torch.stack([data[i+1:i+block_size+1] for i in ix])

# =====================================================================
# Конфигурация
# =====================================================================
@dataclass
class Config:
    data_path: str = "data.txt"
    save_path: str = "tiny_ru.pt"
    block_size: int = 96
    batch_size: int = 64
    emb_dim: int = 96
    hidden_dim: int = 384
    num_layers: int = 1
    dropout: float = 0.1
    lr: float = 3e-3
    steps: int = 2500
    eval_every: int = 250
    sample_every: int = 500
    grad_clip: float = 1.0
    seed: int = 42
    device: str = "cpu"
    prompt: str = "жили-были"


# =====================================================================
# Токенизация
# =====================================================================
def build_vocab(text):
    chars = sorted(set(text))
    stoi = {c: i for i, c in enumerate(chars)}
    itos = {i: c for c, i in stoi.items()}
    return chars, stoi, itos


def encode(text, stoi):
    return torch.tensor([stoi[c] for c in text], dtype=torch.long)


# =====================================================================
# Модель
# =====================================================================
class TinyGRU(nn.Module):
    def __init__(self, vocab_size, emb_dim, hidden_dim, num_layers, dropout):
        super().__init__()
        self.emb = nn.Embedding(vocab_size, emb_dim)
        self.gru = nn.GRU(
            emb_dim, hidden_dim, num_layers,
            batch_first=True,
            dropout=dropout if num_layers > 1 else 0.0,
        )
        self.drop = nn.Dropout(dropout)
        self.fc = nn.Linear(hidden_dim, vocab_size)

    def forward(self, x, h=None):
        e = self.emb(x)
        o, h = self.gru(e, h)
        return self.fc(self.drop(o)), h


# =====================================================================
# Данные
# =====================================================================
def get_batch(data, block_size, batch_size):
    ix = torch.randint(0, len(data) - block_size - 1, (batch_size,))
    x = torch.stack([data[i: i + block_size] for i in ix])
    y = torch.stack([data[i + 1: i + block_size + 1] for i in ix])
    return x, y


# =====================================================================
# Генерация
# =====================================================================
@torch.no_grad()
def generate(model, stoi, itos, prompt, max_new=250,
             temperature=0.8, top_k=10, block_size=96):
    model.eval()
    if not prompt:
        prompt = " "
    ids = [stoi.get(c, 0) for c in prompt]
    x = torch.tensor(ids, dtype=torch.long).unsqueeze(0)
    out = list(ids)

    for _ in range(max_new):
        x_cond = x[:, -block_size:]
        logits, _ = model(x_cond)
        logits = logits[:, -1, :] / temperature
        if top_k is not None:
            v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
            logits[logits < v[:, [-1]]] = -float("inf")
        probs = F.softmax(logits, dim=-1)
        nxt = torch.multinomial(probs, num_samples=1).item()
        out.append(nxt)
        x = torch.cat([x, torch.tensor([[nxt]])], dim=1)

    return "".join(itos[i] for i in out)


# =====================================================================
# Главный цикл
# =====================================================================
def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--data", type=str, default="data.txt")
    p.add_argument("--steps", type=int, default=2500)
    p.add_argument("--block-size", type=int, default=96)
    p.add_argument("--batch-size", type=int, default=64)
    p.add_argument("--emb-dim", type=int, default=96)
    p.add_argument("--hidden", type=int, default=384)
    p.add_argument("--layers", type=int, default=1)
    p.add_argument("--lr", type=float, default=3e-3)
    p.add_argument("--dropout", type=float, default=0.1)
    p.add_argument("--prompt", type=str, default="жили-были")
    p.add_argument("--save", type=str, default="tiny_ru.pt")
    return p.parse_args()


def main():
    args = parse_args()
    cfg = Config(
        data_path=args.data,
        save_path=args.save,
        steps=args.steps,
        block_size=args.block_size,
        batch_size=args.batch_size,
        emb_dim=args.emb_dim,
        hidden_dim=args.hidden,
        num_layers=args.layers,
        lr=args.lr,
        dropout=args.dropout,
        prompt=args.prompt,
    )

    random.seed(cfg.seed)
    torch.manual_seed(cfg.seed)

    print("=" * 60)
    print("  Tiny RU — быстрая нейросеть на CPU")
    print("=" * 60)
    print(f"CPU ядер: {torch.get_num_threads()}")
    print(f"Устройство: {cfg.device}")

    # 1. Текст
    if os.path.exists(cfg.data_path):
        with open(cfg.data_path, "r", encoding="utf-8") as f:
            text = f.read()
        print(f"Загружен файл: {cfg.data_path} ({len(text):,} символов)")
    else:
        text = BUILTIN_TEXT
        print(f"Файл {cfg.data_path} не найден. Использую встроенный текст "
              f"({len(text):,} символов)")
        print("  → для лучшего результата положите рядом data.txt")

    text = text.lower().replace("ё", "е")   # упрощаем: ё → е
    print(f"После нормализации: {len(text):,} символов")

    # 2. Словарь
    chars, stoi, itos = build_vocab(text)
    vocab_size = len(chars)
    print(f"Словарь: {vocab_size} символов")

    # 3. Данные
    data = encode(text, stoi)
    n = len(data)
    n_val = max(cfg.block_size + 1, n // 10)
    train_data = data[: n - n_val]
    val_data = data[n - n_val:]
    print(f"Train: {len(train_data):,} | Val: {len(val_data):,}")

    # 4. Модель
    model = TinyGRU(
        vocab_size=vocab_size,
        emb_dim=cfg.emb_dim,
        hidden_dim=cfg.hidden_dim,
        num_layers=cfg.num_layers,
        dropout=cfg.dropout,
    ).to(cfg.device)

    n_params = sum(p.numel() for p in model.parameters())
    print(f"Параметров: {n_params:,}")
    print("-" * 60)

    optimizer = torch.optim.AdamW(model.parameters(), lr=cfg.lr,
                                  weight_decay=1e-4)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(
        optimizer, T_max=cfg.steps, eta_min=cfg.lr * 0.1
    )

    # 5. Обучение
    t0 = time.time()
    best_val = float("inf")
    running = 0.0

    for step in range(1, cfg.steps + 1):
        model.train()
        x, y = get_batch(train_data, cfg.block_size, cfg.batch_size)
        x, y = x.to(cfg.device), y.to(cfg.device)

        optimizer.zero_grad()
        logits, _ = model(x)
        loss = F.cross_entropy(logits.view(-1, vocab_size), y.view(-1))
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), cfg.grad_clip)
        optimizer.step()
        scheduler.step()

        running += loss.item()

        # оценка и семпл
        if step % cfg.eval_every == 0:
            model.eval()
            with torch.no_grad():
                total, cnt = 0.0, 0
                for _ in range(20):
                    xv, yv = get_batch(val_data, cfg.block_size, cfg.batch_size)
                    lv, _ = model(xv)
                    total += F.cross_entropy(
                        lv.view(-1, vocab_size), yv.view(-1)
                    ).item()
                    cnt += 1
                val_loss = total / cnt
            avg = running / cfg.eval_every
            running = 0.0
            dt = time.time() - t0
            marker = ""
            if val_loss < best_val:
                best_val = val_loss
                torch.save(
                    {"model_state": model.state_dict(),
                     "stoi": stoi, "itos": itos,
                     "chars": chars,
                     "config": cfg.__dict__},
                    cfg.save_path,
                )
                marker = " ✓"

            print(
                f"шаг {step:>5}/{cfg.steps} | "
                f"loss {avg:.3f} | val {val_loss:.3f} | "
                f"ppl {math.exp(val_loss):5.2f} | "
                f"{dt:5.0f}с{marker}"
            )

        if step % cfg.sample_every == 0:
            sample = generate(
                model, stoi, itos,
                prompt=cfg.prompt,
                max_new=150,
                temperature=0.8,
                top_k=10,
                block_size=cfg.block_size,
            )
            print(f"  ─── образец ───")
            print(f"  {sample}")
            print()

    print("-" * 60)
    print(f"Обучение завершено за {time.time() - t0:.1f}с")
    print(f"Лучший val_loss: {best_val:.4f} | ppl {math.exp(best_val):.2f}")
    print(f"Веса сохранены: {cfg.save_path}")

    # 6. Финальная генерация
    print()
    print("=" * 60)
    print("  Генерация")
    print("=" * 60)

    prompts = [cfg.prompt, "дед", "нейросеть", "в лесу", "русский"]
    for p in prompts:
        out = generate(
            model, stoi, itos,
            prompt=p,
            max_new=250,
            temperature=0.75,
            top_k=8,
            block_size=cfg.block_size,
        )
        print(f"\n▶ {p!r}")
        print(out)
        print()


if __name__ == "__main__":
    main()

🧠 Почему именно так
GRU вместо LSTM/Transformer — на CPU GRU обучается быстрее всего при малом размере.

Char-level вместо BPE — не нужен отдельный токенизатор, словарь ~80 символов вместо 8000.

Random sampling батчей — не тратим время на перебор эпох.

torch.set_num_threads — используем все ядра CPU.

Cosine LR — быстрая сходимость за 2–3 тысячи шагов.

Периодическая генерация — сразу видно, учится ли модель.

⚠️ Ограничения
На 6 КБ текста модель переобучится и будет почти дословно повторять корпус.

Осмысленная генерация начинается примерно с 100 КБ – 1 МБ текста.

На 16 ГБ ОЗУ спокойно поместится корпус 50–100 МБ — этого достаточно для приличной маленькой модели.

Если хотите совсем без переобучения — увеличьте --dropout 0.2 и уменьшите --hidden 256.

"""
tiny_model.py — модель + загрузка/сохранение + генерация.
Используется в chat_tiny.py и web_tiny.py.
"""

import torch
import torch.nn as nn
import torch.nn.functional as F


class TinyGRU(nn.Module):
    def __init__(self, vocab_size, emb_dim, hidden_dim, num_layers, dropout):
        super().__init__()
        self.emb = nn.Embedding(vocab_size, emb_dim)
        self.gru = nn.GRU(
            emb_dim, hidden_dim, num_layers,
            batch_first=True,
            dropout=dropout if num_layers > 1 else 0.0,
        )
        self.drop = nn.Dropout(dropout)
        self.fc = nn.Linear(hidden_dim, vocab_size)

    def forward(self, x, h=None):
        e = self.emb(x)
        o, h = self.gru(e, h)
        return self.fc(self.drop(o)), h


def load_tiny(path, device="cpu"):
    """Загружает чекпоинт и возвращает (model, stoi, itos, cfg)."""
    ckpt = torch.load(path, map_location=device)
    cfg = ckpt["config"]
    stoi = ckpt["stoi"]
    itos = ckpt["itos"]

    model = TinyGRU(
        vocab_size=len(stoi),
        emb_dim=cfg["emb_dim"],
        hidden_dim=cfg["hidden_dim"],
        num_layers=cfg["num_layers"],
        dropout=cfg["dropout"],
    ).to(device)
    model.load_state_dict(ckpt["model_state"])
    model.eval()
    return model, stoi, itos, cfg


def apply_top_k_top_p(logits, top_k=None, top_p=None, temperature=1.0):
    """Фильтры для генерации: top-k, top-p (nucleus)."""
    logits = logits / max(temperature, 1e-6)

    if top_k is not None and top_k > 0:
        v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
        logits[logits < v[..., [-1]]] = -float("inf")

    if top_p is not None and 0.0 < top_p < 1.0:
        sorted_logits, sorted_idx = torch.sort(logits, descending=True, dim=-1)
        probs = F.softmax(sorted_logits, dim=-1)
        cum = probs.cumsum(dim=-1)
        mask = cum - probs > top_p
        sorted_logits[mask] = -float("inf")
        # восстанавливаем порядок
        logits = torch.full_like(logits, -float("inf"))
        logits.scatter_(-1, sorted_idx, sorted_logits)

    return logits


@torch.no_grad()
def stream_generate(model, stoi, itos, prompt, max_new=200,
                    temperature=0.8, top_k=10, top_p=None,
                    repetition_penalty=1.1, block_size=96,
                    stop_chars="\n\n"):
    """
    Генератор: отдаёт по одному символу (или по кусочку),
    чтобы UI мог показывать «печатает».
    """
    if not prompt:
        prompt = " "

    ids = [stoi.get(c, 0) for c in prompt.lower()]
    x = torch.tensor(ids, dtype=torch.long).unsqueeze(0)
    out = list(ids)

    buf = ""
    for _ in range(max_new):
        x_cond = x[:, -block_size:]
        logits, _ = model(x_cond)
        logits = logits[:, -1, :]

        # repetition penalty
        if repetition_penalty and repetition_penalty != 1.0:
            seen = set(x[0].tolist()[-64:])
            for tok in seen:
                if logits[0, tok] > 0:
                    logits[0, tok] /= repetition_penalty
                else:
                    logits[0, tok] *= repetition_penalty

        logits = apply_top_k_top_p(logits, top_k=top_k, top_p=top_p,
                                   temperature=temperature)
        probs = F.softmax(logits, dim=-1)
        nxt = torch.multinomial(probs, num_samples=1).item()

        out.append(nxt)
        x = torch.cat([x, torch.tensor([[nxt]])], dim=1)

        ch = itos[nxt]
        buf += ch
        yield ch

        # стоп по абзацу
        if stop_chars and buf.endswith(stop_chars):
            break


def full_generate(model, stoi, itos, prompt, **kw):
    """Обычная (не стриминговая) генерация — возвращает строку."""
    out = []
    for ch in stream_generate(model, stoi, itos, prompt, **kw):
        out.append(ch)
    return "".join(out)
📄 chat_tiny.py — стриминговая генерация в терминале
Работает как ChatGPT в консоли: текст появляется по буквам.

python
"""
chat_tiny.py — интерактивная генерация в терминале.

Запуск:
    python chat_tiny.py
    python chat_tiny.py --model tiny_ru.pt --temp 0.8 --top-k 8
"""

import sys
import time
import argparse

import torch

from tiny_model import load_tiny, stream_generate


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--model", type=str, default="tiny_ru.pt")
    p.add_argument("--temp", type=float, default=0.8)
    p.add_argument("--top-k", type=int, default=10)
    p.add_argument("--top-p", type=float, default=None)
    p.add_argument("--rep", type=float, default=1.15)
    p.add_argument("--tokens", type=int, default=200)
    p.add_argument("--delay", type=float, default=0.01,
                   help="пауза между символами (сек), 0 = мгновенно")
    return p.parse_args()


def print_stream(text, delay=0.01):
    """Печатает по символу с задержкой — эффект 'печатает'."""
    for ch in text:
        sys.stdout.write(ch)
        sys.stdout.flush()
        if delay > 0:
            time.sleep(delay)
    print()


def main():
    args = parse_args()

    print("Загружаю модель...")
    model, stoi, itos, cfg = load_tiny(args.model)
    block_size = cfg["block_size"]
    print(f"Модель: {args.model}")
    print(f"Словарь: {len(stoi)} символов | block_size: {block_size}")
    print()
    print("Введите начало текста и нажмите Enter.")
    print("Пустая строка — выход. Ctrl+C — прервать генерацию.")
    print("-" * 60)

    while True:
        try:
            prompt = input("\n>>> ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nПока!")
            break

        if not prompt:
            print("Выход.")
            break

        print("-" * 60)

        # сам промпт печатаем сразу
        sys.stdout.write(prompt)
        sys.stdout.flush()

        t0 = time.time()
        n_chars = 0
        try:
            for ch in stream_generate(
                model, stoi, itos, prompt,
                max_new=args.tokens,
                temperature=args.temp,
                top_k=args.top_k,
                top_p=args.top_p,
                repetition_penalty=args.rep,
                block_size=block_size,
                stop_chars="\n\n",
            ):
                sys.stdout.write(ch)
                sys.stdout.flush()
                n_chars += 1
                if args.delay > 0:
                    time.sleep(args.delay)
        except KeyboardInterrupt:
            print("\n[прервано]")
            continue

        dt = time.time() - t0
        speed = n_chars / max(dt, 1e-6)
        print()
        print(f"[{n_chars} симв. за {dt:.2f}с — {speed:.1f} симв/с]")


if __name__ == "__main__":
    main()

📄 prepare_data.py — собрать data.txt из папки с текстами
Если у вас есть несколько .txt или .fb2 файлов, склеиваем в один корпус.

python
"""
prepare_data.py — собирает data.txt из папки с текстами.

Запуск:
    python prepare_data.py --input ./books --output data.txt
    python prepare_data.py --input ./books --output data.txt --max-mb 50
"""

import os
import re
import argparse
import unicodedata


def clean_text(text: str, lower: bool = True, keep_yo: bool = True) -> str:
    text = unicodedata.normalize("NFC", text)
    if lower:
        text = text.lower()
    if not keep_yo:
        text = text.replace("ё", "е")
    # убираем html-теги (на случай fb2)
    text = re.sub(r"<[^>]+>", " ", text)
    # схлопываем пробелы, но сохраняем абзацы
    text = re.sub(r"[ \t]+", " ", text)
    text = re.sub(r"\n{3,}", "\n\n", text)
    return text.strip()


def iter_files(folder):
    exts = (".txt", ".fb2", ".md")
    for root, _, files in os.walk(folder):
        for f in files:
            if f.lower().endswith(exts):
                yield os.path.join(root, f)


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--input", type=str, required=True)
    p.add_argument("--output", type=str, default="data.txt")
    p.add_argument("--max-mb", type=int, default=100)
    p.add_argument("--no-lower", action="store_true")
    p.add_argument("--no-yo", action="store_true",
                   help="заменять ё на е")
    return p.parse_args()


def main():
    args = parse_args()
    max_bytes = args.max_mb * 1024 * 1024
    written = 0
    n_files = 0

    print(f"Источник: {args.input}")
    print(f"Выход:    {args.output} (лимит {args.max_mb} МБ)")

    with open(args.output, "w", encoding="utf-8") as out:
        for path in iter_files(args.input):
            try:
                with open(path, "r", encoding="utf-8", errors="ignore") as f:
                    text = f.read()
            except Exception as e:
                print(f"  пропуск {path}: {e}")
                continue

            text = clean_text(
                text,
                lower=not args.no_lower,
                keep_yo=not args.no_yo,
            )
            if len(text) < 200:
                continue

            out.write(text + "\n\n")
            written += len(text.encode("utf-8"))
            n_files += 1
            print(f"  + {path} ({len(text):,} симв.)")

            if written >= max_bytes:
                print(f"Достигнут лимит {args.max_mb} МБ, останавливаюсь.")
                break

    print(f"\nГотово. Файлов: {n_files}, размер: {written/1024/1024:.2f} МБ")


if __name__ == "__main__":
    main()

Обучение 8000 шагов, корпус 10 МБ	~2.5 ч

⚙️ Что можно докрутить
1. Лучше качество на том же CPU
bash
python tiny_ru.py \
    --steps 8000 \
    --hidden 512 \
    --emb-dim 128 \
    --layers 2 \
    --block-size 128 \

2. Beam search
Сейчас sampling (top-k/top-p). Beam search даст более «правильный», но менее разнообразный текст. Могу добавить в tiny_model.py, если нужно.

3. Сохранение истории чата
chat_tiny.py можно расширить: писать лог в chat_log.txt.

4. REST API

5. Квантизация

# lora.py
"""LoRA-подобные адаптеры для TinyGRU."""

import math
import torch
import torch.nn as nn

class LoRALinear(nn.Module):
    """Linear + низкоранговая поправка: W' = W + (alpha/r) * B @ A."""
    def __init__(self, base, r=16, alpha=32, dropout=0.0):
        super().__init__()
        assert isinstance(base, nn.Linear)
        self.base = base
        for p in self.base.parameters():
            p.requires_grad = False
        self.in_features = base.in_features
        self.out_features = base.out_features
        self.r = r
        self.scale = alpha / r
        self.A = nn.Parameter(torch.empty(r, self.in_features))
        self.B = nn.Parameter(torch.zeros(self.out_features, r))
        nn.init.kaiming_uniform_(self.A, a=math.sqrt(5))
        self.drop = nn.Dropout(dropout) if dropout > 0 else nn.Identity()

    def forward(self, x):
        base = self.base(x)
        delta = self.drop(x) @ self.A.t() @ self.B.t()
        return base + self.scale * delta

    def merge(self):
        """Возвращает обычный Linear с уже влитой LoRA."""
        W = self.base.weight.data + self.scale * (self.B @ self.A)
        b = self.base.bias.data.clone() if self.base.bias is not None else None
        new = nn.Linear(self.in_features, self.out_features, bias=b is not None)
        new.weight.data.copy_(W)
        if b is not None:
            new.bias.data.copy_(b)
        return new


class LoRAEmbedding(nn.Module):
    """Embedding + низкоранговая поправка к таблице эмбеддингов."""
    def __init__(self, base, r=16, alpha=32):
        super().__init__()
        assert isinstance(base, nn.Embedding)
        self.base = base
        for p in self.base.parameters():
            p.requires_grad = False
        self.num = base.num_embeddings
        self.dim = base.embedding_dim
        self.r = r
        self.scale = alpha / r
        self.A = nn.Parameter(torch.zeros(self.num, r))
        self.B = nn.Parameter(torch.zeros(r, self.dim))
        nn.init.normal_(self.A, std=0.01)
        nn.init.zeros_(self.B)

    def forward(self, x):
        base = self.base(x)
        delta = self.A[x] @ self.B
        return base + self.scale * delta


def apply_lora_to_model(model, r=16, alpha=32, target_fc=True, target_emb=True):
    """Оборачивает fc/emb в LoRA и замораживает всё остальное."""
    # сначала морозим всё
    for p in model.parameters():
        p.requires_grad = False

    if target_fc:
        model.fc = LoRALinear(model.fc, r=r, alpha=alpha)
    if target_emb:
        model.emb = LoRAEmbedding(model.emb, r=r, alpha=alpha)
    return model


def lora_state_dict(model):
    """Возвращает только обучаемые (LoRA) параметры."""
    return {k: v for k, v in model.state_dict().items() if "A" in k or "B" in k}

📄 lora.py — LoRA-адаптеры
python
"""
lora.py — LoRA-подобные адаптеры для TinyGRU.

Идея LoRA: замораживаем базовый вес W и учим низкоранговую поправку
    W' = W + (alpha/r) * B @ A
где A: (r, in), B: (out, r), r << min(in, out).
"""

import math
import torch
import torch.nn as nn


class LoRALinear(nn.Module):
    """Linear + низкоранговая поправка."""
    def __init__(self, base: nn.Linear, r: int = 16, alpha: int = 32,
                 dropout: float = 0.0):
        super().__init__()
        assert isinstance(base, nn.Linear)
        self.base = base
        for p in self.base.parameters():
            p.requires_grad = False

        self.in_features = base.in_features
        self.out_features = base.out_features
        self.r = r
        self.scale = alpha / r

        # B инициализируем нулями — на старте LoRA-поправка = 0
        self.A = nn.Parameter(torch.empty(r, self.in_features))
        self.B = nn.Parameter(torch.zeros(self.out_features, r))
        nn.init.kaiming_uniform_(self.A, a=math.sqrt(5))

        self.drop = nn.Dropout(dropout) if dropout > 0 else nn.Identity()

    def forward(self, x):
        base = self.base(x)
        delta = self.drop(x) @ self.A.t() @ self.B.t()
        return base + self.scale * delta

    @torch.no_grad()
    def merged(self) -> nn.Linear:
        """Возвращает обычный Linear, в который влита LoRA."""
        W = self.base.weight.data + self.scale * (self.B @ self.A)
        b = self.base.bias.data.clone() if self.base.bias is not None else None
        new = nn.Linear(self.in_features, self.out_features, bias=b is not None)
        new.weight.data.copy_(W)
        if b is not None:
            new.bias.data.copy_(b)
        return new


class LoRAEmbedding(nn.Module):
    """Embedding + низкоранговая поправка к таблице эмбеддингов."""
    def __init__(self, base: nn.Embedding, r: int = 16, alpha: int = 32):
        super().__init__()
        assert isinstance(base, nn.Embedding)
        self.base = base
        for p in self.base.parameters():
            p.requires_grad = False

        self.num_embeddings = base.num_embeddings
        self.embedding_dim = base.embedding_dim
        self.r = r
        self.scale = alpha / r

        self.A = nn.Parameter(torch.zeros(self.num_embeddings, r))
        self.B = nn.Parameter(torch.zeros(r, self.embedding_dim))
        nn.init.normal_(self.A, std=0.01)
        nn.init.zeros_(self.B)

    def forward(self, x):
        base = self.base(x)
        delta = self.A[x] @ self.B
        return base + self.scale * delta

    @torch.no_grad()
    def merged(self) -> nn.Embedding:
        W = self.base.weight.data + self.scale * (self.A @ self.B)
        new = nn.Embedding(self.num_embeddings, self.embedding_dim)
        new.weight.data.copy_(W)
        return new

# Утилиты
# =====================================================================
def apply_lora_to_model(model, r=16, alpha=32,
                        target_emb=True, target_fc=True):
    """
    Замораживает все параметры модели и оборачивает emb/fc в LoRA.
    Модель модифицируется на месте и возвращается.
    """
    # 1. морозим всё
    for p in model.parameters():
        p.requires_grad = False

    # 2. заменяем слои на LoRA-обёртки
    if target_emb and isinstance(getattr(model, "emb", None), nn.Embedding):
        model.emb = LoRAEmbedding(model.emb, r=r, alpha=alpha)
    if target_fc and isinstance(getattr(model, "fc", None), nn.Linear):
        model.fc = LoRALinear(model.fc, r=r, alpha=alpha)
    return model


def trainable_state(model):
    """Только обучаемые параметры (LoRA)."""
    return {n: p.data.clone()
            for n, p in model.named_parameters() if p.requires_grad}


def load_trainable_state(model, state):
    """Загружает обучаемые параметры."""
    with torch.no_grad():
        for n, p in model.named_parameters():
            if p.requires_grad and n in state:
                p.data.copy_(state[n])


def count_trainable(model):
    return sum(p.numel() for p in model.parameters() if p.requires_grad)


@torch.no_grad()
def merge_lora_inplace(model):
    """
    Вливает LoRA обратно в базовые веса и убирает обёртки.
    После этого модель — обычный TinyGRU без LoRA.
    """
    if isinstance(getattr(model, "emb", None), LoRAEmbedding):
        model.emb = model.emb.merged()
    if isinstance(getattr(model, "fc", None), LoRALinear):
        model.fc = model.fc.merged()
    return model

"""
chat_style.py — интерактивный чат со стилизованной моделью.

Запуск:
    python chat_style.py --adapter style_adapter.pt
    python chat_style.py --adapter style_merged.pt       # слитая

import sys
import time
import argparse

import torch

from tiny_model import load_tiny, stream_generate
from lora import (
    apply_lora_to_model, load_trainable_state,
)


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--base", type=str, default="tiny_ru.pt")
    p.add_argument("--adapter", type=str, default="style_adapter.pt")
    p.add_argument("--temp", type=float, default=0.75)
    p.add_argument("--top-k", type=int, default=8)
    p.add_argument("--rep", type=float, default=1.15)
    p.add_argument("--tokens", type=int, default=200)
    p.add_argument("--delay", type=float, default=0.01)
    return p.parse_args()


def load_styled_model(base_path, adapter_path):
    """Собирает модель: база + LoRA-адаптер (или слитая)."""
    adapter = torch.load(adapter_path, map_location="cpu")

    # Если это слитая модель — просто загружаем как обычную
    if adapter.get("merged"):
        model, stoi, itos, cfg = load_tiny(adapter_path, device="cpu")
        return model, stoi, itos, cfg, "merged"

    # Иначе — база + LoRA
    model, stoi, itos, cfg = load_tiny(base_path, device="cpu")

    mode = adapter.get("mode", "lora")
    if mode == "lora":
        model = apply_lora_to_model(
            model,
            r=adapter["lora_r"],
            alpha=adapter["lora_alpha"],
        )
        load_trainable_state(model, adapter["adapter"])
    else:
        # full fine-tune — все веса в адаптере
        sd = model.state_dict()
        for k, v in adapter["adapter"].items():
            if k in sd:
                sd[k].copy_(v)
        model.load_state_dict(sd)

    return model, stoi, itos, cfg, mode


def main():
    args = parse_args()

    print("Загружаю базовую модель + адаптер...")
    model, stoi, itos, cfg, mode = load_styled_model(args.base, args.adapter)
    model.eval()

    print(f"База:       {args.base}")
    print(f"Адаптер:    {args.adapter} ({mode})")
    print(f"Словарь:    {len(stoi)} символов")
    print()
    print("Введите промпт, Enter — генерация, пусто — выход.")
    print("-" * 60)

    while True:
        try:
            prompt = input("\n>>> ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nПока!")
            break
        if not prompt:
            break

        sys.stdout.write(prompt)
        sys.stdout.flush()
        n_chars = 0
        t0 = time.time()
        try:
            for ch in stream_generate(
                model, stoi, itos, prompt,
                max_new=args.tokens,
                temperature=args.temp,
                top_k=args.top_k,
                repetition_penalty=args.rep,
                block_size=cfg["block_size"],
                stop_chars="\n\n",
            ):
                sys.stdout.write(ch)
                sys.stdout.flush()
                n_chars += 1
                if args.delay > 0:
                    time.sleep(args.delay)
        except KeyboardInterrupt:
            print("\n[прервано]")
            continue

        dt = time.time() - t0
        print()
        print(f"[{n_chars} симв. за {dt:.2f}с — "
              f"{n_chars / max(dt, 1e-6):.1f} симв/с]")


if __name__ == "__main__":
    main()

🧠 Как это работает
1. Что такое LoRA здесь
Обычный Linear: y = W·x + b.
LoRA: y = W·x + b + (α/r)·B·A·x,
где W заморожен, а A и B — маленькие обучаемые матрицы.

Для fc в TinyGRU:

W: 384 → 78 (29 952 параметра)

A: 16 × 384 = 6 144

B: 78 × 16 = 1 248

Итого LoRA: 7 392 вместо 29 952 (в 4 раза меньше)

Для emb:

W: 78 × 96 = 7 488

A: 78 × 16 = 1 248

B: 16 × 96 = 1 536

Итого: 2 784

Всего обучаемых: ~20 000 из 335 000 = 6 %.

2. Почему это работает
Стиль — это низкоразмерный сигнал. Не нужно менять все веса, чтобы сдвинуть распределение к «робот-стилю».

LoRA учит именно сдвиг в пространстве признаков, не перезаписывая базовые знания.

Меньше параметров → быстрее сходится, не переобучается на маленьком корпусе.

Не разрушает базу — можно держать один tiny_ru.pt и десяток адаптеров под разные стили.

3. Что обучается, а что нет
Компонент	Заморожен	Учится

GRU (все веса)	✅	❌
fc.weight, fc.bias	✅	❌
fc.A, fc.B (LoRA)	❌	✅

📄 models_zoo.py — зоопарк моделей
Пять архитектур разной мощности. У всех единый интерфейс forward(x, h=None) -> (logits, h).

python
"""
models_zoo.py — разные архитектуры для одной задачи:
    предсказание следующего символа.

Все модели:
    forward(x) -> logits формы (B, T, vocab_size)
У bigram — своя сигнатура, см. ниже.
"""

import math
import torch
import torch.nn as nn
import torch.nn.functional as F


# =====================================================================
# 1. Bigram — таблица переходов. Обучение мгновенное.
# =====================================================================
class BigramModel(nn.Module):
    """
    Простейшая модель: P(next | current) — матрица vocab x vocab.
    Логиты = строка таблицы, индекс по текущему символу.
    """
    def __init__(self, vocab_size, **kwargs):
        super().__init__()
        self.table = nn.Embedding(vocab_size, vocab_size)
        nn.init.zeros_(self.table.weight)

    def forward(self, x, h=None):
        return self.table(x), None


# =====================================================================
# 2. MLP на контексте фиксированной длины
# =====================================================================
class MLPModel(nn.Module):
    """
    Берёт последние `context` символов, конкатенирует эмбеддинги,
    прогоняет через MLP, предсказывает следующий символ.
    Очень быстрый, но контекст ограничен.
    """
    def __init__(self, vocab_size, emb_dim=64, hidden_dim=256,
                 num_layers=1, dropout=0.1, context=8, **kwargs):
        super().__init__()
        self.context = context
        self.emb = nn.Embedding(vocab_size, emb_dim)
        layers = []
        in_dim = emb_dim * context
        for _ in range(num_layers):
            layers += [
                nn.Linear(in_dim, hidden_dim),
                nn.GELU(),
                nn.Dropout(dropout),
            ]
            in_dim = hidden_dim
        layers.append(nn.Linear(in_dim, vocab_size))
        self.net = nn.Sequential(*layers)

    def forward(self, x, h=None):
        B, T = x.shape
        # паддим слева, чтобы можно было применять ко всей последовательности
        if T < self.context:
            pad = torch.zeros(B, self.context - T, dtype=x.dtype,
                              device=x.device)
            x = torch.cat([pad, x], dim=1)
            T = self.context

        e = self.emb(x)                       # (B, T, E)
        outs = []
        for t in range(T):
            start = max(0, t - self.context + 1)
            ctx = e[:, start:t + 1]           # (B, <=context, E)
            if ctx.size(1) < self.context:
                pad = torch.zeros(B, self.context - ctx.size(1),
                                  e.size(-1), device=e.device)
                ctx = torch.cat([pad, ctx], dim=1)
            outs.append(self.net(ctx.reshape(B, -1)))
        return torch.stack(outs, dim=1), None


# =====================================================================
# 3. GRU
# =====================================================================
class GRUModel(nn.Module):
    def __init__(self, vocab_size, emb_dim=96, hidden_dim=384,
                 num_layers=1, dropout=0.1, **kwargs):
        super().__init__()
        self.emb = nn.Embedding(vocab_size, emb_dim)
        self.rnn = nn.GRU(
            emb_dim, hidden_dim, num_layers,
            batch_first=True,
            dropout=dropout if num_layers > 1 else 0.0,
        )
        self.drop = nn.Dropout(dropout)
        self.fc = nn.Linear(hidden_dim, vocab_size)

    def forward(self, x, h=None):
        e = self.emb(x)
        o, h = self.rnn(e, h)
        return self.fc(self.drop(o)), h


# =====================================================================
# 4. LSTM
# =====================================================================
class LSTMModel(nn.Module):
    def __init__(self, vocab_size, emb_dim=96, hidden_dim=384,
                 num_layers=1, dropout=0.1, **kwargs):
        super().__init__()
        self.emb = nn.Embedding(vocab_size, emb_dim)
        self.rnn = nn.LSTM(
            emb_dim, hidden_dim, num_layers,
            batch_first=True,
            dropout=dropout if num_layers > 1 else 0.0,
        )
        self.drop = nn.Dropout(dropout)
        self.fc = nn.Linear(hidden_dim, vocab_size)

    def forward(self, x, h=None):
        e = self.emb(x)
        o, h = self.rnn(e, h)
        return self.fc(self.drop(o)), h


# =====================================================================
# 5. Mini-Transformer (decoder-only)
# =====================================================================
class CausalSelfAttention(nn.Module):
    def __init__(self, n_embd, n_head, dropout, block_size):
        super().__init__()
        assert n_embd % n_head == 0
        self.n_head = n_head
        self.head_dim = n_embd // n_head
        self.qkv = nn.Linear(n_embd, 3 * n_embd, bias=False)
        self.proj = nn.Linear(n_embd, n_embd, bias=False)
        self.drop = nn.Dropout(dropout)
        self.register_buffer(
            "mask",
            torch.tril(torch.ones(block_size, block_size)).view(
                1, 1, block_size, block_size
            ),
            persistent=False,
        )

    def forward(self, x):
        B, T, C = x.shape
        q, k, v = self.qkv(x).chunk(3, dim=-1)
        q = q.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        k = k.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        v = v.view(B, T, self.n_head, self.head_dim).transpose(1, 2)
        att = (q @ k.transpose(-2, -1)) / math.sqrt(self.head_dim)
        att = att.masked_fill(self.mask[:, :, :T, :T] == 0, float("-inf"))
        att = F.softmax(att, dim=-1)
        att = self.drop(att)
        out = (att @ v).transpose(1, 2).contiguous().view(B, T, C)
        return self.drop(self.proj(out))


class TransformerBlock(nn.Module):
    def __init__(self, n_embd, n_head, dropout, block_size):
        super().__init__()
        self.ln1 = nn.LayerNorm(n_embd)
        self.attn = CausalSelfAttention(n_embd, n_head, dropout, block_size)
        self.ln2 = nn.LayerNorm(n_embd)
        self.mlp = nn.Sequential(
            nn.Linear(n_embd, 4 * n_embd),
            nn.GELU(),
            nn.Linear(4 * n_embd, n_embd),
            nn.Dropout(dropout),
        )

    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        x = x + self.mlp(self.ln2(x))
        return x


class TransformerModel(nn.Module):
    def __init__(self, vocab_size, emb_dim=96, n_head=4, num_layers=2,
                 dropout=0.1, block_size=96, **kwargs):
        super().__init__()
        self.block_size = block_size
        self.tok_emb = nn.Embedding(vocab_size, emb_dim)
        self.pos_emb = nn.Embedding(block_size, emb_dim)
        self.drop = nn.Dropout(dropout)
        self.blocks = nn.ModuleList([
            TransformerBlock(emb_dim, n_head, dropout, block_size)
            for _ in range(num_layers)
        ])
        self.ln_f = nn.LayerNorm(emb_dim)
        self.fc = nn.Linear(emb_dim, vocab_size, bias=False)
        self.fc.weight = self.tok_emb.weight

    def forward(self, x, h=None):
        B, T = x.shape
        pos = torch.arange(T, device=x.device).unsqueeze(0)
        h = self.drop(self.tok_emb(x) + self.pos_emb(pos))
        for block in self.blocks:
            h = block(h)
        return self.fc(self.ln_f(h)), None


# =====================================================================
# Реестр моделей
# =====================================================================
MODEL_REGISTRY = {
    "bigram":      BigramModel,
    "mlp":         MLPModel,
    "gru":         GRUModel,
    "lstm":        LSTMModel,
    "transformer": TransformerModel,
}


def build_model(name, vocab_size, **kwargs):
    if name not in MODEL_REGISTRY:
        raise ValueError(f"Неизвестная модель: {name}. "
                         f"Доступно: {list(MODEL_REGISTRY)}")
    return MODEL_REGISTRY[name](vocab_size=vocab_size, **kwargs)

@dataclass
class Candidate:
    name: str
    arch: str
    kwargs: Dict[str, Any] = field(default_factory=dict)
    lr: float = 3e-3
    batch_size: int = 64
    block_size: int = 96
    steps: int = 400


# Список кандидатов по умолчанию. Порядок — от слабых к сильным.
DEFAULT_CANDIDATES: List[Candidate] = [

# =====================================================================
# Данные
# =====================================================================
def build_vocab(text):
    chars = sorted(set(text))
    stoi = {c: i for i, c in enumerate(chars)}
    itos = {i: c for c, i in stoi.items()}
    return chars, stoi, itos


def encode(text, stoi):
    return torch.tensor([stoi[c] for c in text], dtype=torch.long)


def get_batch(data, block_size, batch_size):
    ix = torch.randint(0, len(data) - block_size - 1, (batch_size,))
    x = torch.stack([data[i: i + block_size] for i in ix])
    y = torch.stack([data[i + 1: i + block_size + 1] for i in ix])
    return x, y

# Обучение одного кандидата
# =====================================================================
@dataclass
class Result:
    name: str
    arch: str
    kwargs: Dict[str, Any]
    train_loss: float
    val_loss: float
    ppl: float
    seconds: float
    params: int
    trainable: int
    status: str = "ok"
    note: str = ""


def train_candidate(cand: Candidate, train_data, val_data,
                    vocab_size, device="cpu", verbose=True):
    """Обучает одного кандидата, возвращает Result + обученную модель."""
    torch.manual_seed(42)
    random.seed(42)

    t0 = time.time()
    try:
        model = build_model(cand.arch, vocab_size, **cand.kwargs).to(device)
    except Exception as e:
        return Result(
            name=cand.name, arch=cand.arch, kwargs=cand.kwargs,
            train_loss=float("inf"), val_loss=float("inf"), ppl=float("inf"),
            seconds=0.0, params=0, trainable=0,
            status="build_error", note=str(e),
        ), None

    n_params = sum(p.numel() for p in model.parameters())
    n_train = sum(p.numel() for p in model.parameters() if p.requires_grad)

    optimizer = torch.optim.AdamW(
        model.parameters(), lr=cand.lr, weight_decay=1e-4
    )
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(
        optimizer, T_max=max(1, cand.steps), eta_min=cand.lr * 0.1
    )

    running = 0.0
    try:
        for step in range(1, cand.steps + 1):
            model.train()
            x, y = get_batch(train_data, cand.block_size, cand.batch_size)
            x, y = x.to(device), y.to(device)

            optimizer.zero_grad()
            logits, _ = model(x)
            loss = F.cross_entropy(
                logits.view(-1, vocab_size), y.view(-1)
            )
            loss.backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
            optimizer.step()
            scheduler.step()
            running += loss.item()

            if verbose and step % max(1, cand.steps // 4) == 0:
                print(f"    шаг {step:>4}/{cand.steps} | "
                      f"loss {running / step:.4f}")
    except Exception as e:
        return Result(
            name=cand.name, arch=cand.arch, kwargs=cand.kwargs,
            train_loss=float("inf"), val_loss=float("inf"), ppl=float("inf"),
            seconds=time.time() - t0,
            params=n_params, trainable=n_train,
            status="train_error", note=str(e),
        ), None

    train_loss = running / cand.steps

    # валидация
    model.eval()
    with torch.no_grad():
        total, cnt = 0.0, 0
        for _ in range(20):
            xv, yv = get_batch(val_data, cand.block_size, cand.batch_size)
            xv, yv = xv.to(device), yv.to(device)
            lv, _ = model(xv)
            total += F.cross_entropy(
                lv.view(-1, vocab_size), yv.view(-1)
            ).item()
            cnt += 1
        val_loss = total / cnt

    return Result(
        name=cand.name, arch=cand.arch, kwargs=cand.kwargs,
        train_loss=train_loss, val_loss=val_loss,
        ppl=math.exp(val_loss),
        seconds=time.time() - t0,
        params=n_params, trainable=n_train,
    ), model

# Печать таблицы результатов
# =====================================================================
def print_results(results: List[Result]):
    print()
    print("=" * 96)
    print(f"{'модель':<16}{'архит.':<14}{'params':>10}"
          f"{'train':>10}{'val':>10}{'ppl':>10}{'время':>10}{'статус':>10}")
    print("-" * 96)
    for r in sorted(results, key=lambda x: x.val_loss):
        print(f"{r.name:<16}{r.arch:<14}{r.params:>10,}"
              f"{r.train_loss:>10.4f}{r.val_loss:>10.4f}"
              f"{r.ppl:>10.2f}{r.seconds:>9.1f}с"
              f"{r.status:>10}")
    print("=" * 96)

# Ансамбль: усредняем логиты топ-K моделей
# =====================================================================
class EnsembleModel(nn.Module):
    """Усредняет вероятности нескольких моделей."""
    def __init__(self, models, weights=None):
        super().__init__()
        self.models = nn.ModuleList(models)
        if weights is None:
            weights = [1.0 / len(models)] * len(models)
        self.weights = weights

    @torch.no_grad()
    def forward(self, x, h=None):
        total = None
        for m, w in zip(self.models, self.weights):
            logits, _ = m(x)
            probs = F.softmax(logits, dim=-1)
            total = probs * w if total is None else total + probs * w
        return torch.log(total + 1e-9), None


def evaluate_ensemble(ensemble, val_data, block_size, batch_size, vocab_size):
    ensemble.eval()
    with torch.no_grad():
        total, cnt = 0.0, 0
        for _ in range(20):
            x, y = get_batch(val_data, block_size, batch_size)
            logits, _ = ensemble(x)
            total += F.cross_entropy(
                logits.view(-1, vocab_size), y.view(-1)
            ).item()
            cnt += 1
    return total / cnt

def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--data", type=str, default="data.txt")
    p.add_argument("--steps", type=int, default=None,
                   help="переопределить число шагов у всех кандидатов")
    p.add_argument("--budget", type=int, default=None,
                   help="лимит времени в секундах на весь поиск")
    p.add_argument("--top-k", type=int, default=3,
                   help="сколько лучших моделей использовать в ансамбле")
    p.add_argument("--ensemble", action="store_true",
                   help="построить ансамбль из топ-K")
    p.add_argument("--save", type=str, default="best_model.pt")
    p.add_argument("--report", type=str, default="search_report.json")
    p.add_argument("--quick", action="store_true",
                   help="быстрый режим: только 3 кандидата")
    return p.parse_args()

    args = parse_args()
    device = "cpu"

    print("=" * 96)
    print("  AUTO-BEST: поиск лучшей модели")
    print("=" * 96)
    print(f"CPU ядер: {torch.get_num_threads()}")

    # 1. Данные

    data = encode(text, stoi)

    val_data = data[len(data) - n_val:]
    print(f"Словарь: {vocab_size} симв. | "

    # 2. Кандидаты
    candidates = DEFAULT_CANDIDATES

        for c in candidates:
            c.steps = args.steps

    print(f"Кандидатов: {len(candidates)}")

    # 3. Поиск
    print("-" * 96)
    results: List[Result] = []
    trained_models: Dict[str, nn.Module] = {}

    t_start = time.time()
    for i, cand in enumerate(candidates, 1):

# Универсальная генерация для любой модели из зоопарка
# =====================================================================
@torch.no_grad()
def generate_simple(model, stoi, itos, prompt, max_new=180,
                    temperature=0.8, top_k=8, block_size=96):
    model.eval()
    if not prompt:
        prompt = " "
    ids = [stoi.get(c, 0) for c in prompt]
    x = torch.tensor(ids, dtype=torch.long).unsqueeze(0)
    out = list(ids)

    for _ in range(max_new):
        x_cond = x[:, -block_size:]
        logits, _ = model(x_cond)
        logits = logits[:, -1, :] / temperature
        if top_k:
            v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
            logits[logits < v[:, [-1]]] = -float("inf")
        probs = F.softmax(logits, dim=-1)
        nxt = torch.multinomial(probs, num_samples=1).item()
        out.append(nxt)
        x = torch.cat([x, torch.tensor([[nxt]])], dim=1)

    return "".join(itos[i] for i in out)


if __name__ == "__main__":
    main()

если ансамбль лучше лучшей модели — сохраняем отдельно.

class Bigram(nn.Module):
    def __init__(self, V, **kw):
        super().__init__()
        self.t = nn.Embedding(V, V)
        nn.init.zeros_(self.t.weight)

    def forward(self, x, h=None):
        return self.t(x), None


class MLP(nn.Module):
    def __init__(self, V, emb_dim=64, hidden_dim=256, context=8, dropout=0.1, **kw):
        super().__init__()
        self.ctx = context
        self.emb = nn.Embedding(V, emb_dim)
        self.net = nn.Sequential(
            nn.Linear(emb_dim * context, hidden_dim), nn.GELU(), nn.Dropout(dropout),
            nn.Linear(hidden_dim, hidden_dim), nn.GELU(), nn.Dropout(dropout),
            nn.Linear(hidden_dim, V),
        )

    def forward(self, x, h=None):
        B, T = x.shape
        if T < self.ctx:
            pad = torch.zeros(B, self.ctx - T, dtype=x.dtype, device=x.device)
            x = torch.cat([pad, x], 1); T = self.ctx
        e = self.emb(x)
        outs = []
        for t in range(T):
            s = max(0, t - self.ctx + 1)
            c = e[:, s:t + 1]
            if c.size(1) < self.ctx:
                c = torch.cat([torch.zeros(B, self.ctx - c.size(1), e.size(-1), device=e.device), c], 1)
            outs.append(self.net(c.reshape(B, -1)))
        return torch.stack(outs, 1), None


class GRU(nn.Module):
    def __init__(self, V, emb_dim=96, hidden_dim=384, num_layers=1, dropout=0.1, **kw):
        super().__init__()
        self.emb = nn.Embedding(V, emb_dim)
        self.rnn = nn.GRU(emb_dim, hidden_dim, num_layers, batch_first=True,
                          dropout=dropout if num_layers > 1 else 0.0)
        self.drop = nn.Dropout(dropout)
        self.fc = nn.Linear(hidden_dim, V)

    def forward(self, x, h=None):
        o, h = self.rnn(self.emb(x), h)
        return self.fc(self.drop(o)), h


class LSTM(nn.Module):
    def __init__(self, V, emb_dim=96, hidden_dim=384, num_layers=1, dropout=0.1, **kw):
        super().__init__()
        self.emb = nn.Embedding(V, emb_dim)
        self.rnn = nn.LSTM(emb_dim, hidden_dim, num_layers, batch_first=True,
                           dropout=dropout if num_layers > 1 else 0.0)
        self.drop = nn.Dropout(dropout)
        self.fc = nn.Linear(hidden_dim, V)

    def forward(self, x, h=None):
        o, h = self.rnn(self.emb(x), h)
        return self.fc(self.drop(o)), h


class Attn(nn.Module):
    def __init__(self, d, nh, dp, bs):
        super().__init__()
        self.nh, self.hd = nh, d // nh
        self.qkv = nn.Linear(d, 3 * d, bias=False)
        self.proj = nn.Linear(d, d, bias=False)
        self.dp = nn.Dropout(dp)
        self.register_buffer("mask", torch.tril(torch.ones(bs, bs)).view(1, 1, bs, bs), persistent=False)

    def forward(self, x):
        B, T, C = x.shape
        q, k, v = self.qkv(x).chunk(3, -1)
        q = q.view(B, T, self.nh, self.hd).transpose(1, 2)
        k = k.view(B, T, self.nh, self.hd).transpose(1, 2)
        v = v.view(B, T, self.nh, self.hd).transpose(1, 2)
        a = (q @ k.transpose(-2, -1)) / math.sqrt(self.hd)
        a = a.masked_fill(self.mask[:, :, :T, :T] == 0, float("-inf"))
        a = self.dp(F.softmax(a, -1))
        return self.proj((a @ v).transpose(1, 2).contiguous().view(B, T, C))


class Block(nn.Module):
    def __init__(self, d, nh, dp, bs):
        super().__init__()
        self.ln1, self.ln2 = nn.LayerNorm(d), nn.LayerNorm(d)
        self.attn = Attn(d, nh, dp, bs)
        self.mlp = nn.Sequential(nn.Linear(d, 4 * d), nn.GELU(), nn.Linear(4 * d, d), nn.Dropout(dp))

    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        x = x + self.mlp(self.ln2(x))
        return x


class Transformer(nn.Module):
    def __init__(self, V, emb_dim=128, n_head=4, num_layers=3, dropout=0.1, block_size=128, **kw):
        super().__init__()
        self.bs = block_size
        self.tok = nn.Embedding(V, emb_dim)
        self.pos = nn.Embedding(block_size, emb_dim)
        self.drop = nn.Dropout(dropout)
        self.blocks = nn.ModuleList([Block(emb_dim, n_head, dropout, block_size) for _ in range(num_layers)])
        self.ln_f = nn.LayerNorm(emb_dim)
        self.head = nn.Linear(emb_dim, V, bias=False)
        self.head.weight = self.tok.weight
        self.apply(self._init)

    def _init(self, m):
        if isinstance(m, nn.Linear):
            nn.init.normal_(m.weight, 0.0, 0.02)
            if m.bias is not None: nn.init.zeros_(m.bias)
        elif isinstance(m, nn.Embedding):
            nn.init.normal_(m.weight, 0.0, 0.02)

    def forward(self, x, h=None):
        B, T = x.shape
        p = torch.arange(T, device=x.device).unsqueeze(0)
        z = self.drop(self.tok(x) + self.pos(p))
        for b in self.blocks: z = b(z)
        return self.head(self.ln_f(z)), None

# ---------- 7. TRAINING ----------
@dataclass
class TrialResult:
    name: str
    arch: str
    seed: int
    best_val: float
    best_step: int
    train_loss: float
    val_loss: float
    ppl: float
    seconds: float
    params: int
    stopped_early: bool
    history: List[Tuple[int, float, float]]


@torch.no_grad()
def evaluate(model: nn.Module, data: torch.Tensor, block: int, batch: int, V: int, n_batches: int, gen: torch.Generator) -> float:
    model.eval()
    tot = 0.0
    for _ in range(n_batches):
        x, y = get_batch(data, block, batch, gen)
        logits, _ = model(x)
        tot += F.cross_entropy(logits.view(-1, V), y.view(-1)).item()
    return tot / n_batches


def train_trial(cand: Candidate, train: torch.Tensor, val: torch.Tensor, V: int, seed: int, cfg: Config) -> Tuple[TrialResult, nn.Module]:
    set_seed(seed)
    model = REGISTRY[cand.arch](V, **cand.kwargs)
    n_params = sum(p.numel() for p in model.parameters())
    opt = torch.optim.AdamW(model.parameters(), lr=cand.lr, weight_decay=cfg.weight_decay)
    sch = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=cand.max_steps, eta_min=cand.lr * 0.1)
    gen = torch.Generator().manual_seed(seed)

    best_val, best_state, best_step, wait, stopped = float("inf"), None, 0, 0, False
    running, history = 0.0, []
    t0 = time.time()

    for step in range(1, cand.max_steps + 1):
        model.train()
        x, y = get_batch(train, cand.block_size, cand.batch_size, gen)
        opt.zero_grad()
        logits, _ = model(x)
        loss = F.cross_entropy(logits.view(-1, V), y.view(-1))
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), cfg.grad_clip)
        opt.step()
        sch.step()
        running += loss.item()

        if step % cand.eval_every == 0:
            vl = evaluate(model, val, cand.block_size, cand.batch_size, V, cfg.eval_batches, gen)
            tr = running / cand.eval_every
            running = 0.0
            history.append((step, tr, vl))

            if vl < best_val - cfg.min_delta:
                best_val, best_step, wait = vl, step, 0
                best_state = {k: v.detach().clone() for k, v in model.state_dict().items()}
            else:
                wait += 1
                if wait >= cfg.patience:
                    stopped = True
                    break

    if best_state is not None:
        model.load_state_dict(best_state)

    return TrialResult(
        name=cand.name, arch=cand.arch, seed=seed,
        best_val=best_val, best_step=best_step,
        train_loss=history[-1][1] if history else float("inf"),
        val_loss=best_val, ppl=math.exp(best_val),
        seconds=time.time() - t0, params=n_params,
        stopped_early=stopped, history=history,
    ), model


# ---------- 8. STATISTICS ----------
def mean_ci(xs: List[float], conf: float = 0.95) -> Tuple[float, float, float]:
    n = len(xs)
    if n < 2:
        return (xs[0] if xs else float("nan"), float("nan"), float("nan"))
    m = statistics.mean(xs)
    sd = statistics.stdev(xs)
    se = sd / math.sqrt(n)
    z = 1.96 if conf == 0.95 else 2.576
    return m, m - z * se, m + z * se


def paired_t(xs: List[float], ys: List[float]) -> Tuple[float, float]:
    if len(xs) != len(ys) or len(xs) < 2:
        return float("nan"), float("nan")
    d = [a - b for a, b in zip(xs, ys)]
    md = statistics.mean(d)
    sd = statistics.stdev(d)
    if sd == 0:
        return float("inf") if md != 0 else 0.0, 0.0 if md != 0 else 1.0
    t = md / (sd / math.sqrt(len(d)))
    df = len(d) - 1
    return t, df


def aggregate_trials(results: List[TrialResult]) -> Dict[str, Dict[str, Any]]:
    by_name: Dict[str, List[TrialResult]] = {}
    for r in results:
        by_name.setdefault(r.name, []).append(r)

    agg: Dict[str, Dict[str, Any]] = {}
    for name, trials in by_name.items():
        vals = [t.val_loss for t in trials]
        m, lo, hi = mean_ci(vals)
        agg[name] = {
            "arch": trials[0].arch,
            "n_seeds": len(trials),
            "mean_val": m,
            "ci_low": lo,
            "ci_high": hi,
            "std_val": statistics.stdev(vals) if len(vals) > 1 else 0.0,
            "mean_ppl": math.exp(m),
            "mean_sec": statistics.mean(t.seconds for t in trials),
            "params": trials[0].params,
            "seeds": [t.seed for t in trials],
            "vals": vals,
        }
    return agg


# ---------- 9. MODEL SELECTION ----------
def select_best(agg: Dict[str, Dict[str, Any]], top_k: int = 3) -> List[str]:
    return [n for n, _ in sorted(agg.items(), key=lambda kv: kv[1]["mean_val"])[:top_k]]


def confidence_rank(agg: Dict[str, Dict[str, Any]]) -> List[Tuple[str, float, float, float]]:
    rows = []
    for n, s in agg.items():
        rows.append((n, s["mean_val"], s["ci_low"], s["ci_high"]))
    return sorted(rows, key=lambda r: r[1])


# ---------- 10. FINAL TEST ----------
@torch.no_grad()
def final_test(model: nn.Module, test: torch.Tensor, block: int, batch: int, V: int, n_batches: int = 64) -> Tuple[float, float, float]:
    gen = torch.Generator().manual_seed(0)
    model.eval()
    tot, n = 0.0, 0
    correct, total = 0, 0
    for _ in range(n_batches):
        x, y = get_batch(test, block, batch, gen)
        logits, _ = model(x)
        tot += F.cross_entropy(logits.view(-1, V), y.view(-1)).item()
        n += 1
        pred = logits.argmax(-1)
        correct += (pred == y).sum().item()
        total += y.numel()
    return tot / n, math.exp(tot / n), correct / total


# ---------- 11. GENERATION ----------
@torch.no_grad()
def generate(model: nn.Module, stoi: Dict[str, int], itos: Dict[int, str],
             prompt: str, max_new: int = 200, temp: float = 0.8,
             top_k: int = 10, block: int = 128) -> str:
    model.eval()
    ids = [stoi.get(c, 0) for c in (prompt or " ")]
    x = torch.tensor(ids, dtype=torch.long).unsqueeze(0)
    out = list(ids)
    for _ in range(max_new):
        logits, _ = model(x[:, -block:])
        logits = logits[:, -1, :] / temp
        if top_k:
            v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
            logits[logits < v[:, [-1]]] = -float("inf")
        nxt = torch.multinomial(F.softmax(logits, -1), 1).item()
        out.append(nxt)
        x = torch.cat([x, torch.tensor([[nxt]])], 1)
    return "".join(itos[i] for i in out)


# ---------- 12. PERSISTENCE ----------
def save_artifact(path: str, model: nn.Module, cand: Candidate, V: int,
                  stoi, itos, metrics: Dict[str, float]) -> None:
    torch.save({
        "state": model.state_dict(),
        "arch": cand.arch,
        "kwargs": cand.kwargs,
        "block_size": cand.block_size,
        "vocab_size": V,
        "stoi": stoi, "itos": itos,
        "metrics": metrics,
    }, path)


def save_report(path: str, agg: Dict[str, Any], results: List[TrialResult],
                winner: str, test_metrics: Dict[str, float]) -> None:
    with open(path, "w", encoding="utf-8") as f:
        json.dump({
            "aggregate": agg,
            "trials": [asdict(r) for r in results],
            "winner": winner,
            "test": test_metrics,
        }, f, ensure_ascii=False, indent=2, default=str)


# ---------- 13. MAIN ----------
def parse_args() -> argparse.Namespace:
    p = argparse.ArgumentParser()
    p.add_argument("--data", type=str, default="data.txt")
    p.add_argument("--out", type=str, default="runs")
    p.add_argument("--steps", type=int, default=None)
    p.add_argument("--seeds", type=int, nargs="+", default=[42, 1337, 2024])
    p.add_argument("--budget", type=int, default=None)
    p.add_argument("--top-k", type=int, default=3)
    p.add_argument("--quick", action="store_true")
    return p.parse_args()


def main() -> None:
    args = parse_args()
    cfg = Config(data_path=args.data, save_dir=args.out, seeds=tuple(args.seeds))
    os.makedirs(cfg.save_dir, exist_ok=True)
    set_seed(cfg.seeds[0])

    print("=" * 100)
    print("  SCIENTIFIC MODEL SELECTION · Russian char-level LM · CPU")
    print("=" * 100)
    print(f"threads={torch.get_num_threads()} · device={cfg.device} · seeds={cfg.seeds}")

    # 1. Data
    text = normalize(load_corpus(cfg.data_path))
    chars, stoi, itos = build_vocab(text)
    V = len(chars)
    data = encode(text, stoi)
    train, val, test = split_data(data, cfg.val_ratio, cfg.test_ratio)
    print(f"corpus={len(text):,} · vocab={V} · train={len(train):,} · val={len(val):,} · test={len(test):,}")
    print(f"sha256={sha256(text)}")

    # 2. Candidates
    candidates = default_candidates()
    if args.quick:
        keep = {"bigram", "mlp-c8", "gru-m", "tf-tiny"}
        candidates = [c for c in candidates if c.name in keep]
    if args.steps is not None:
        for c in candidates:
            c.max_steps = args.steps
    print(f"candidates={len(candidates)} · seeds_per_candidate={len(cfg.seeds)}")

    # 3. Trials
    print("-" * 100)
    results: List[TrialResult] = []
    trained: Dict[Tuple[str, int], nn.Module] = {}
    t_start = time.time()

    for ci, cand in enumerate(candidates, 1):
        if args.budget and (time.time() - t_start) > args.budget:
            print(f"[budget] stop at candidate {ci}")
            break
        for seed in cfg.seeds:
            print(f"[{ci}/{len(candidates)}] {cand.name:<10} seed={seed:<5}", end=" ", flush=True)
            r, m = train_trial(cand, train, val, V, seed, cfg)
            results.append(r)
            trained[(cand.name, seed)] = m
            print(f"val={r.val_loss:.4f} ppl={r.ppl:.2f} step={r.best_step} "
                  f"{r.seconds:.1f}s early={r.stopped_early}")

    # 4. Aggregation
    agg = aggregate_trials(results)
    ranked = confidence_rank(agg)

    print()
    print("=" * 100)
    print(f"{'rank':<5}{'model':<12}{'arch':<13}{'params':>9}"
          f"{'mean_val':>12}{'95% CI':>22}{'std':>9}{'ppl':>9}{'sec':>8}")
    print("-" * 100)
    for i, (name, m, lo, hi) in enumerate(ranked, 1):
        s = agg[name]
        ci = f"[{lo:.4f}, {hi:.4f}]"
        print(f"{i:<5}{name:<12}{s['arch']:<13}{s['params']:>9,}"
              f"{m:>12.4f}{ci:>22}{s['std_val']:>9.4f}{s['mean_ppl']:>9.2f}{s['mean_sec']:>8.1f}")
    print("=" * 100)

    # 5. Winner
    top = select_best(agg, top_k=args.top_k)
    winner = top[0]
    print(f"winner={winner} · arch={agg[winner]['arch']} · "
          f"mean_val={agg[winner]['mean_val']:.4f} · "
          f"CI=[{agg[winner]['ci_low']:.4f}, {agg[winner]['ci_high']:.4f}]")

    # 6. Paired t-test vs runner-up
    if len(top) >= 2:
        a, b = top[0], top[1]
        t, df = paired_t(agg[a]["vals"], agg[b]["vals"])
        print(f"paired_t({a} vs {b}) = {t:.3f} · df={df}")

    # 7. Retrain winner on train+val, evaluate on test
    print("-" * 100)
    print(f"retraining winner={winner} on train+val, testing on held-out test")
    best_cand = next(c for c in candidates if c.name == winner)
    best_cand.max_steps = int(best_cand.max_steps * 1.15)
    train_val = torch.cat([train, val])
    _, final_model = train_trial(best_cand, train_val, test, V, cfg.seeds[0], cfg)
    test_loss, test_ppl, test_acc = final_test(final_model, test, best_cand.block_size, best_cand.batch_size, V)
    print(f"test_loss={test_loss:.4f} · test_ppl={test_ppl:.2f} · test_acc={test_acc:.4f}")

    # 8. Save
    save_artifact(os.path.join(cfg.save_dir, "best_model.pt"),
                  final_model, best_cand, V, stoi, itos,
                  {"val_loss": agg[winner]["mean_val"],
                   "test_loss": test_loss, "test_ppl": test_ppl, "test_acc": test_acc})
    save_report(os.path.join(cfg.save_dir, "search_report.json"),
                agg, results, winner,
                {"test_loss": test_loss, "test_ppl": test_ppl, "test_acc": test_acc})
    print(f"saved: {cfg.save_dir}/best_model.pt · {cfg.save_dir}/search_report.json")

    # 9. Sample
    print("=" * 100)
    print("SAMPLE")
    print("=" * 100)
    for prompt in ["жили-были", "в лесу", "нейросеть"]:
        out = generate(final_model, stoi, itos, prompt, max_new=180,
                       temp=0.8, top_k=10, block=best_cand.block_size)
        print(f"▶ {prompt!r}\n{out}\n")

    # 10. Summary
    print("=" * 100)
    print(f"total_time={time.time() - t_start:.1f}s · "
          f"trials={len(results)} · candidates={len(candidates)} · "
          f"winner={winner} · test_ppl={test_ppl:.2f}")
    print("=" * 100)


if __name__ == "__main__":
    main()
