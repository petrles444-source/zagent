  <script>
    /*
     * ============================================================
     *  Ruffle Advanced Template
     *  ------------------------------------------------------------
     *  1. Глобальная конфигурация Ruffle
     *  2. Ленивая инициализация при появлении в viewport

    // URL SWF-файла — замените на свой

    // Путь к Ruffle. Можно использовать CDN:
    const RUFFLE_CDN = 'https://unpkg.com/@ruffle-rs/ruffle';
    // Или self-hosted (скачайте ruffle-selfhosted и укажите путь):
    // const RUFFLE_CDN = '/path/to/ruffle/ruffle.js';

    // Глобальные настройки Ruffle (применяются ко всем плеерам на странице)
    window.RufflePlayer = window.RufflePlayer || {};
    window.RufflePlayer.config = {
      // --- Общие настройки страницы ---
      publicPath: undefined,          // путь к ресурсам Ruffle (если self-hosted)
      polyfills: true,                // автоматически заменять <object>/<embed>

      // --- Настройки конкретного файла ---
      autoplay: 'on',                 // 'on' | 'auto' | 'off' — автозапуск
      unmuteOverlay: 'hidden',        // 'visible' | 'hidden' — скрыть оверлей «включить звук»
      splashScreen: false,            // убрать заставку Ruffle
      backgroundColor: '#000000',     // цвет фона плеера
      letterbox: 'fullscreen',        // 'fullscreen' | 'on' | 'off' — чёрные полосы
      scale: 'showAll',               // 'showAll' | 'exactFit' | 'noBorder' | 'noScale'
      quality: 'high',                // 'low' | 'medium' | 'high' | 'best'
      wmode: 'opaque',                // 'window' | 'opaque' | 'transparent'
      allowScriptAccess: true,        // разрешить ActionScript ↔ JS
      allowNetworking: 'all',         // 'all' | 'internal' | 'none'
      allowFullscreen: true,          // разрешить полноэкранный режим
      contextMenu: 'on',              // 'on' | 'off' | 'rightClick'
      showSwfDownload: false,         // скрыть пункт «Скачать SWF» в контекстном меню
      logLevel: 'error',              // 'error' | 'warn' | 'info' | 'debug' | 'trace'
      maxExecutionDuration: 15,       // макс. время выполнения ActionScript (сек)
      upgradeToHttps: true,           // автоматически переводить HTTP → HTTPS
      compatibilityRules: true,       // включить правила совместимости
      favorFlash: false,              // отдавать приоритет поведению Flash
      warnOnUnsupportedContent: true, // предупреждать о неподдерживаемом контенте
      preloader: true,                // показывать встроенный прелоадер
      forceAlign: false,              // принудительное выравнивание
      salign: '',                     // выравнивание сцены
      fullScreenAspectRatio: 'showAll',
      frameRate: null,                // null = из SWF
      preferredRenderer: 'webgpu',    // 'webgpu' | 'canvas' | 'webgl'
      base: null,                     // базовый URL для относительных путей
      menu: true                      // показывать встроенное меню Ruffle

    /** Показать статус в бейдже */
    function setStatus(text, autoHide = false) {
      statusBadge.textContent = text;
      statusBadge.classList.remove('is-hidden');

    /** Загрузить скрипт Ruffle динамически */
    function loadRuffleScript() {
      return new Promise((resolve, reject) => {
        if (window.RufflePlayer && window.RufflePlayer.newest) {
          resolve(); // уже загружен
          return;
        }
        const script = document.createElement('script');
        script.src = RUFFLE_CDN;
        script.async = true;
        script.onload = () => {
          // Даём Ruffle время инициализироваться
          const check = setInterval(() => {
            if (window.RufflePlayer && window.RufflePlayer.newest) {
              clearInterval(check);
              resolve();
            }
          }, 50);
          // Таймаут на инициализацию
          setTimeout(() => {
            clearInterval(check);
            if (!window.RufflePlayer || !window.RufflePlayer.newest) {
              reject(new Error('Ruffle не инициализировался'));
            }
          }, 5000);
        };
        script.onerror = () => reject(new Error('Не удалось загрузить Ruffle'));
        document.head.appendChild(script);
      });
    }

    async function initPlayer() {
      try {
        setStatus('Загрузка Ruffle…');
        await loadRuffleScript();

        setStatus('Создание плеера…');
        const ruffle = window.RufflePlayer.newest();
        player = ruffle.createPlayer();

        player.config = {

          unmuteOverlay: 'hidden',
          splashScreen: false,
          backgroundColor: '#000000',
          scale: 'showAll',

          allowScriptAccess: true,
          allowFullscreen: true,
          letterbox: 'fullscreen',
          logLevel: 'error',
          showSwfDownload: false,

        // Добавляем плеер в DOM (внутри обёртки, поверх фона)
        stage.appendChild(player);
        // Скрываем fallback, если он был виден
        fallback.hidden = true;

        // Загружаем SWF

        // Начальная синхронизация громкости
        volumeSlider.value = player.volume;

        // Подписываемся на события плеера (если поддерживаются)
        // Ruffle не предоставляет стандартных DOM-событий вроде 'play',
        // но мы можем отслеживать readyState и metadata.
        // Для простоты используем опрос readyState.

        );
      }

    // Это экономит ресурсы, если на странице несколько плееров.
    let initialized = false;

    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting && !initialized) {
          initialized = true;
          observer.disconnect();
          initPlayer();
        }
      });
    }, {
      rootMargin: '200px', // начинаем загрузку за 200px до появления
      threshold: 0.01
    });

    observer.observe(stage);

    // ---------- 5. УПРАВЛЕНИЕ ВОСПРОИЗВЕДЕНИЕМ ----------

      if (!player) return;
      if (player.suspended) {
        player.resume();
        btnPlayPause.textContent = '⏸ Пауза';
      } else {
        player.suspend();
        btnPlayPause.textContent = '▶ Воспроизвести';
      }
    });

    btnRestart.addEventListener('click', async () => {
      if (!player) return;
      try {
        await player.reload();
        setStatus('Перезапуск…', true);
        // После reload плеер сохраняет настройки, но нужно сбросить кнопку
        btnPlayPause.textContent = '⏸ Пауза';
      } catch (e) {
        console.error('[Ruffle] Ошибка перезапуска:', e);
      }
    });

    volumeSlider.addEventListener('input', () => {
      if (player) {
        player.volume = parseFloat(volumeSlider.value);
      }
    });

    btnFullscreen.addEventListener('click', () => {
      if (!player) return;
      if (player.isFullscreen) {
        player.exitFullscreen();
      } else {
        player.requestFullscreen();
      }
    });

    // ---------- 6. ДОПОЛНИТЕЛЬНО: ГЛОБАЛЬНЫЙ ОБРАБОТЧИК ОШИБОК ----------

    window.addEventListener('error', (event) => {
      // Если ошибка связана с Ruffle, показываем fallback
      if (event.message && event.message.includes('ruffle')) {
        console.error('[Ruffle] Глобальная ошибка:', event.message);
      }
    });

    // ---------- 7. ЭКСПОРТ ДЛЯ ОТЛАДКИ ----------
    // В консоли можно писать: window.__rufflePlayer для доступа к плееру
    Object.defineProperty(window, '__rufflePlayer', {
      get() { return player; }
    });

Что делает этот шаблон «умным»
Глобальная + локальная конфигурация.
window.RufflePlayer.config задаёт defaults для всех плееров, а player.config переопределяет их для конкретного экземпляра. Это соответствует официальной архитектуре Ruffle.

Программное создание через JS API.
Вместо автоматического полифилла <object> используется window.RufflePlayer.newest().createPlayer() и player.load(). Это даёт полный контроль и позволяет обрабатывать промисы загрузки.

Ленивая инициализация.
IntersectionObserver запускает загрузку Ruffle и SWF только когда контейнер приближается к экрану. Это снижает нагрузку и ускоряет первоначальную отрисовку страницы.

Панель управления.
Кнопки паузы/возобновления (suspend()/resume()), перезапуска (reload()), слайдер громкости (volume) и полный экран (requestFullscreen()/exitFullscreen()) используют нативный API PlayerV1.

Fallback и обработка ошибок.
Если Ruffle не загрузился или SWF недоступен, показывается сообщение со ссылкой на скачивание файла. Таймауты защищают от «зависания» на этапе инициализации.

Адаптивность.
Контейнер использует aspect-ratio с запасным вариантом через padding-bottom для старых браузеров.

Отладка.
Через window.__rufflePlayer можно получить доступ к экземпляру плеера из консоли.

Как адаптировать
Замените SWF_URL на путь к вашему файлу.

Ruffle: полная документация, общее резюме и правила
1. Что такое Ruffle
Ruffle — это эмулятор Adobe Flash Player, написанный на языке Rust. Он работает как на десктопе, так и в браузере через WebAssembly, воспроизводя SWF-файлы без необходимости установки оригинального плагина Flash. Ruffle использует современные веб-технологии (HTML5, WebAssembly, WebGL) и обеспечивает безопасность благодаря песочнице браузера и гарантиям безопасности памяти Rust, избегая уязвимостей, которыми славился Flash Player.

Проект распространяется под двойной лицензией MIT и Apache 2.0. Ruffle доступен для Windows, macOS, Linux, а также в виде браузерных расширений для Chrome и Firefox.

4. Веб-интеграция
4.1. Способы развёртывания
Ruffle можно развернуть на сайте несколькими способами, каждый из которых использует общий пакет ruffle-core, но отличается механизмом инициализации.

Способ	Назначение
CDN	Одна строка кода, никакой дополнительной работы
Self-hosted	Собственный хостинг файлов Ruffle
NPM	Программная интеграция в JavaScript-приложения
Браузерное расширение	Автоматическая инъекция Ruffle на любые сайты
4.2. Подключение через CDN (одна строка)
Самый простой способ — добавить скрипт Ruffle на страницу. Ruffle автоматически найдёт и заменит Flash-контент (теги <object> и <embed>).

html
<script src="https://unpkg.com/@ruffle-rs/ruffle"></script>
4.3. Self-hosted пакет
Пакет ruffle-selfhosted предназначен для сайтов, которые не используют сборщики и npm. Он поставляется в виде единого JavaScript-бандла, который можно подключить через тег <script>.

Пакет:

Регистрирует себя как источник Ruffle через Setup.installRuffle("local")

Автоматически вызывает polyfill() для обнаружения и замены Flash-контента

Загружает WASM-модуль из той же директории, что и скрипт

Важно: файл .wasm должен отдаваться с правильным MIME-типом. Если возникает ошибка Incorrect response MIME type, настройте сервер соответствующим образом.

4.4. NPM-пакет
NPM-пакет позволяет программно интегрировать Ruffle в JavaScript-приложения:

javascript
import { Setup } from "ruffle-core";
const ruffle = Setup.installRuffle("local").newest();
const player = ruffle.createPlayer();
document.getElementById("container").appendChild(player);
await player.load({ url: "movie.swf" });
NPM-пакет экспортирует тот же API, что и self-hosted бандл, но позволяет использовать tree-shaking и интеграцию с современными сборщиками (Webpack, Vite).

4.5. Архитектура веб-интеграции
Ruffle использует версионированный API для обеспечения совместимости с различными хост-средами. Публичный API доступен через window.RufflePlayer и предоставляет метод newest() для получения последней загруженной версии движка.

Ядро веб-интеграции — WebAssembly-модуль, скомпилированный из Rust. Модуль загружается динамически на основе возможностей браузера:

Обнаружение функций: проверка расширений WebAssembly

Динамический импорт: загрузка соответствующего WASM-модуля

Отслеживание прогресса: опциональная отчётность через ReadableStream

Инициализация: вызов init() из wasm-bindgen

Мемоизация: кэширование загруженных конструкторов

Мост Rust/JavaScript использует wasm-bindgen для двунаправленного взаимодействия.

Основной интерфейс — BaseLoadOptions, который применяется к window.RufflePlayer.config для установки конфигурации по умолчанию для всех плееров на странице.

window.RufflePlayer.config = {

  publicPath: undefined,
  polyfills: true,
  
  // Опции, влияющие на файлы
  autoplay: "on",
  unmuteOverlay: "hidden",
  backgroundColor: "#000000",
  letterbox: "fullscreen",
  warnOnUnsupportedContent: true,
  contextMenu: false,
  showSwfDownload: false,
  upgradeToHttps: window.location.protocol === "https:",
  maxExecutionDuration: { secs: 15, nanos: 0 },
  logLevel: "error",
  base: null,
  menu: true,
  salign: "",
  scale: "showAll",
  quality: "medium"
};

9. Правила использования Ruffle
Правило 1: Выбирайте правильный способ развёртывания
Для быстрого прототипа → CDN (одна строка)

Для продакшена → self-hosted пакет (контроль над версией)

Для сложных приложений → NPM-пакет

Для конечных пользователей → браузерное расширение

Правило 2: Всегда указывайте fallback
Если Ruffle не загрузился или SWF не поддерживается, покажите пользователю сообщение со ссылкой на скачивание файла. Никогда не оставляйте пустой экран.

Правило 3: Настраивайте MIME-типы
Файл .wasm должен отдаваться с MIME-типом application/wasm. Без этого Ruffle не загрузится.

Правило 4: Используйте версионированный API
Всегда используйте player.ruffle(1) или PlayerV1 вместо прямого доступа к PlayerElement. Прямые методы устарели и могут быть удалены.

Правило 5: Управляйте автозапуском
Не используйте autoplay: "on" без необходимости. Браузеры блокируют автовоспроизведение со звуком. Для интерактивного контента используйте autoplay: "auto" (запуск по клику).

Правило 6: Ограничивайте выполнение ActionScript
Устанавливайте maxExecutionDuration (например, 15 секунд), чтобы предотвратить зависание плеера из-за бесконечных циклов в SWF.

Правило 7: Ленивая загрузка для нескольких плееров
Если на странице несколько SWF, используйте IntersectionObserver для инициализации плеера только при появлении в зоне видимости. Это экономит ресурсы.

Правило 8: Проверяйте совместимость контента
Перед интеграцией проверьте SWF на совместимость с Ruffle. Контент на ActionScript 3.0 может работать не полностью.

Правило 9: Обновляйте Ruffle
Ruffle активно развивается. Регулярно обновляйте версию, чтобы получать исправления и новые возможности.

Правило 10: Соблюдайте лицензию
Ruffle распространяется под MIT и Apache 2.0. При использовании в коммерческих проектах сохраняйте уведомления об авторских правах.
